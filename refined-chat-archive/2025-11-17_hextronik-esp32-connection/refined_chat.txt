# Quick summary / checklist
1. **Identify power pins** on the board: `5V` (or `S5V`) and `GND`.  
2. **Use a common ground.** Connect board GND to ESP32 GND.  
3. **Powering:** power the hexTronik board with 5 V (if its silk says `5V` / `S5V`) and power the ESP32 from its normal source (USB or 5V VIN). Don’t forget common ground.  
4. **Signal connections:** connect the channel outputs (AIL, ELE, THR, RUDD, AUX or OUT1..OUT8) to ESP32 GPIOs. If the board outputs servo-style PWM (typical), measure/assume 3.3–5 V pulses. If they are 5 V pulses, use a level shifter or a simple series resistor + clamp to protect the ESP32.  
5. **Read signals in software** by measuring pulse width (1000–2000 µs typical) with `pulseIn()` or interrupts.

---

# Wiring (practical example)
Assume the board pin labels shown on the PCB are correct.

Hardware:
- `HEX_GND` → ESP32 `GND`
- `HEX_5V`  → 5 V supply (or ESP32 `VIN` if you power ESP32 from same 5 V regulator)
- `AIL`     → ESP32 `GPIO 16` (example)
- `ELE`     → ESP32 `GPIO 17`
- `THR`     → ESP32 `GPIO 18`
- `RUDD`    → ESP32 `GPIO 19`
- `AUX`     → ESP32 `GPIO 21`

If you use `OUT1..OUT8` instead, map them to whichever free GPIOs you want.

Important: if channel outputs are **5 V** pulses, place a level shifter or a simple voltage divider to bring them to 3.3 V (or use a 74HC14/74HCT/logic-level translator). A safe quick option: series 4.7k resistor + 3.3 V clamp (e.g., BZT52 diode) — but level shifter is preferable.

---

# How to detect signal type (quick non-destructive tests)
1. With the board powered, measure voltage between its `5V` pad and `GND` — confirm ~5.0 V.  
2. Probe a channel pin (AIL/THR) while whatever transmitter is running or while the board is powered; you should see pulses. Use a multimeter in frequency mode or an oscilloscope.  
3. If pulses go up to ~5 V, plan level shifting.

---

# Minimal example Arduino-style code (ESP32, reading PWM with pulseIn)
```cpp
// Example: read throttle PWM on GPIO 18
const int THR_PIN = 18;

void setup() {
  Serial.begin(115200);
  pinMode(THR_PIN, INPUT);
}

void loop() {
  // Wait for HIGH pulse and measure width (microseconds)
  unsigned long width = pulseIn(THR_PIN, HIGH, 30000); // timeout 30ms
  if (width > 0) {
    Serial.print("THR pulse: ");
    Serial.println(width); // expect ~1000..2000
  } else {
    Serial.println("No pulse");
  }
  delay(200);
}
```
Notes:
- `pulseIn()` is simple but blocking. For multiple fast channels, use interrupts or RMT peripheral on ESP32 for robust capture.
- Map measured widths to servo angles or command values: `map(width, 1000, 2000, 0, 180)`.

---

# If the board speaks serial or SBUS instead of PWM
Some modern receiver modules output SBUS or a UART. If you measure a continuous serial-like waveform on a channel pin (lots of fast bits), it may be SBUS (inverted UART @100000 bps) or plain UART. If you suspect that:
- Use an oscilloscope or logic analyzer to inspect waveform shape and baud.

```cpp
#include "driver/rmt.h"

#define RMT_CLK_DIV 80   // 80 MHz / 80 = 1 MHz (1 tick = 1 µs)

// GPIO mapping
#define PIN_AIL   25
#define PIN_ELE   26
#define PIN_THR   27
#define PIN_RUDD  14
#define PIN_AUX   12

struct ServoChannel {
  rmt_channel_t channel;
  gpio_num_t pin;
};

ServoChannel servos[5] = {
  {RMT_CHANNEL_0, GPIO_NUM_25}, // AIL
  {RMT_CHANNEL_1, GPIO_NUM_26}, // ELE
  {RMT_CHANNEL_2, GPIO_NUM_27}, // THR
  {RMT_CHANNEL_3, GPIO_NUM_14}, // RUDD
  {RMT_CHANNEL_4, GPIO_NUM_12}  // AUX
};

rmt_item32_t rmt_frame[2];

void setupServo(const ServoChannel& s) {
  rmt_config_t config = {};
  config.rmt_mode = RMT_MODE_TX;
  config.channel = s.channel;
  config.gpio_num = s.pin;
  config.mem_block_num = 1;
  config.tx_config.loop_en = true;
  config.tx_config.carrier_en = false;
  config.tx_config.idle_output_en = true;
  config.tx_config.idle_level = RMT_IDLE_LEVEL_LOW;
  config.clk_div = RMT_CLK_DIV;

  rmt_config(&config);
  rmt_driver_install(s.channel, 0, 0);
}

void setServoPulse(const ServoChannel& s, uint16_t pulse_us) {
  if (pulse_us < 1000) pulse_us = 1000;
  if (pulse_us > 2000) pulse_us = 2000;

  rmt_frame[0].level0 = 1;
  rmt_frame[0].duration0 = pulse_us;
  rmt_frame[0].level1 = 0;
  rmt_frame[0].duration1 = 20000 - pulse_us;

  rmt_write_items(s.channel, rmt_frame, 1, false);
}

void setup() {
  Serial.begin(115200);

  for (int i = 0; i < 5; i++) {
    setupServo(servos[i]);
    setServoPulse(servos[i], 1500); // center all
  }

  setServoPulse(servos[2], 1000); // throttle LOW
}

void loop() {
  // Example: sweep AIL
  static uint16_t pulse = 1000;
  static int dir = 1;

  setServoPulse(servos[0], pulse);

  pulse += dir * 10;
  if (pulse >= 2000 || pulse <= 1000) dir = -dir;

  delay(20); // update rate, NOT timing critical
}
```

```cpp
#define RMT_CLK_DIV 80        // 80 MHz / 80 = 1 MHz → 1 tick = 1 µs
#define SERVO_MIN   1000
#define SERVO_MAX   2000
#define SERVO_SAFE  1000
#define SERVO_FRAME 20000     // 20 ms = 50 Hz

/* ================= GPIO MAP ================= */
#define PIN_AIL   25
#define PIN_ELE   26
#define PIN_THR   27
#define PIN_RUDD  14
#define PIN_AUX   12

/* ================= STRUCT ================= */
struct ServoChannel {
  rmt_channel_t rmt;
  gpio_num_t pin;
};

/* ================= CHANNELS ================= */
ServoChannel servos[5] = {
  {RMT_CHANNEL_0, GPIO_NUM_25}, // AIL
  {RMT_CHANNEL_1, GPIO_NUM_26}, // ELE
  {RMT_CHANNEL_2, GPIO_NUM_27}, // THR
  {RMT_CHANNEL_3, GPIO_NUM_14}, // RUDD
  {RMT_CHANNEL_4, GPIO_NUM_12}  // AUX
};

/* ================= STATE ================= */
uint16_t channelValue[5] = {
  1500, // AIL
  1500, // ELE
  1000, // THR (SAFE)
  1500, // RUDD
  1500  // AUX
};

rmt_item32_t rmtItem;

/* ================= FUNCTIONS ================= */

void setupServo(const ServoChannel& s) {
  rmt_config_t cfg = {};
  cfg.rmt_mode = RMT_MODE_TX;
  cfg.channel = s.rmt;
  cfg.gpio_num = s.pin;
  cfg.mem_block_num = 1;
  cfg.tx_config.loop_en = true;
  cfg.tx_config.carrier_en = false;
  cfg.tx_config.idle_output_en = true;
  cfg.tx_config.idle_level = RMT_IDLE_LEVEL_LOW;
  cfg.clk_div = RMT_CLK_DIV;

  rmt_config(&cfg);
  rmt_driver_install(s.rmt, 0, 0);
}

void setServoPulse(const ServoChannel& s, uint16_t pulse) {
  pulse = constrain(pulse, SERVO_MIN, SERVO_MAX);

  rmtItem.level0 = 1;
  rmtItem.duration0 = pulse;
  rmtItem.level1 = 0;
  rmtItem.duration1 = SERVO_FRAME - pulse;

  rmt_write_items(s.rmt, &rmtItem, 1, false);
}

void emergencyStop() {
  for (int i = 0; i < 5; i++) {
    channelValue[i] = SERVO_SAFE;
    setServoPulse(servos[i], SERVO_SAFE);
  }
}

/* ================= SERIAL ================= */

void processCommand(String cmd) {
  cmd.trim();
  cmd.toUpperCase();

  if (cmd == "STOP") {
    emergencyStop();
    Serial.println("EMERGENCY STOP");
    return;
  }

  int eq = cmd.indexOf('=');
  if (eq < 0) return;

  int value = cmd.substring(eq + 1).toInt();
  value = constrain(value, SERVO_MIN, SERVO_MAX);

  if      (cmd.startsWith("AIL="))  channelValue[0] = value;
  else if (cmd.startsWith("ELE="))  channelValue[1] = value;
  else if (cmd.startsWith("THR="))  channelValue[2] = value;
  else if (cmd.startsWith("RUDD=")) channelValue[3] = value;
  else if (cmd.startsWith("AUX="))  channelValue[4] = value;
}

void handleSerial() {
  static String buffer;

  while (Serial.available()) {
    char c = Serial.read();
    if (c == '\n' || c == '\r') {
      if (buffer.length()) processCommand(buffer);
      buffer = "";
    } else {
      buffer += c;
    }
  }
}

/* ================= SETUP ================= */

void setup() {
  Serial.begin(115200);

  for (int i = 0; i < 5; i++) {
    setupServo(servos[i]);
    setServoPulse(servos[i], channelValue[i]);
  }

  emergencyStop();
  Serial.println("ESP32 RMT SERVO READY");
}

/* ================= LOOP ================= */

void loop() {
  handleSerial();

  for (int i = 0; i < 5; i++) {
    setServoPulse(servos[i], channelValue[i]);
  }

  delay(10); // pacing only, timing handled by RMT
}
```

  {RMT_CHANNEL_2, GPIO_NUM_26}, // THR

```cpp
/* ================= RMT ================= */
void setupServo(const ServoChannel& s) {
  rmt_config_t cfg = {};
  cfg.rmt_mode = RMT_MODE_TX;
  cfg.channel = s.channel;
  cfg.gpio_num = s.pin;
  cfg.mem_block_num = 1;
  cfg.tx_config.loop_en = true;
  cfg.tx_config.carrier_en = false;
  cfg.tx_config.idle_output_en = true;
  cfg.tx_config.idle_level = RMT_IDLE_LEVEL_LOW;
  cfg.clk_div = RMT_CLK_DIV;

  rmt_config(&cfg);
  rmt_driver_install(s.channel, 0, 0);
}

void setServoPulse(const ServoChannel& s, uint16_t pulse) {
  pulse = constrain(pulse, SERVO_MIN, SERVO_MAX);

  rmtItem.level0 = 1;
  rmtItem.duration0 = pulse;
  rmtItem.level1 = 0;
  rmtItem.duration1 = SERVO_FRAME - pulse;

  rmt_write_items(s.channel, &rmtItem, 1, false);
}

/* ================= SAFETY ================= */
void emergencyStop() {
  for (int i = 0; i < 5; i++) {
    channelValue[i] = SERVO_SAFE;
    setServoPulse(servos[i], SERVO_SAFE);
  }
  Serial.println("\n*** EMERGENCY STOP ***");
}

/* ================= SERIAL ================= */
void processCommand(String cmd) {
  cmd.trim();
  cmd.toUpperCase();

  Serial.print("\nCMD: ");
  Serial.println(cmd);

  if (cmd == "STOP") {
    emergencyStop();
    return;
  }

  int eq = cmd.indexOf('=');
  if (eq < 0) {
    Serial.println("ERR FORMAT");
    return;
  }

  int value = cmd.substring(eq + 1).toInt();
  value = constrain(value, SERVO_MIN, SERVO_MAX);

  if      (cmd.startsWith("AIL="))  channelValue[0] = value;
  else if (cmd.startsWith("ELE="))  channelValue[1] = value;
  else if (cmd.startsWith("THR="))  channelValue[2] = value;
  else if (cmd.startsWith("RUDD=")) channelValue[3] = value;
  else if (cmd.startsWith("AUX="))  channelValue[4] = value;
  else {
    Serial.println("ERR UNKNOWN");
    return;
  }

  Serial.println("OK");
}

void handleSerial() {
  static String buffer;

  while (Serial.available()) {
    char c = Serial.read();

    // Echo every character so user KNOWS input arrived
    Serial.write(c);

    // Newline handling (optional)
    if (c == '\n' || c == '\r') {
      if (buffer.length()) {
        processCommand(buffer);
        buffer = "";
      }
    } 
    else {
      buffer += c;

      // Process immediately if command looks complete
      if (buffer.length() >= 5 && buffer.indexOf('=') > 0) {
        int eq = buffer.indexOf('=');
        if (buffer.length() > eq + 1) {
          processCommand(buffer);
          buffer = "";
        }
      }
    }
  }
}

/* ================= SETUP ================= */
void setup() {
  Serial.begin(115200);
  delay(300);

  Serial.println("\nESP32 RMT SERIAL ONLINE");

  for (int i = 0; i < 5; i++) {
    setupServo(servos[i]);
    setServoPulse(servos[i], channelValue[i]);
  }

  emergencyStop();
}

/* ================= LOOP ================= */
void loop() {
  handleSerial();

  for (int i = 0; i < 5; i++) {
    setServoPulse(servos[i], channelValue[i]);
  }

  delay(10);
}
```

```cpp
#define RMT_CLK_DIV 80
#define SERVO_MIN   1000
#define SERVO_MAX   2000
#define SERVO_SAFE  1000
#define SERVO_FRAME 20000
#define SERIAL_TIMEOUT 2000   // ms
#define THR_RAMP_STEP 5        // µs per loop

/* ================= GPIO MAP ================= */
#define PIN_AIL   14
#define PIN_ELE   27
#define PIN_THR   26
#define PIN_RUDD  25
#define PIN_AUX   33

struct ServoChannel {
  rmt_channel_t channel;
  gpio_num_t pin;
};

ServoChannel servos[5] = {
  {RMT_CHANNEL_0, GPIO_NUM_14},
  {RMT_CHANNEL_1, GPIO_NUM_27},
  {RMT_CHANNEL_2, GPIO_NUM_26},
  {RMT_CHANNEL_3, GPIO_NUM_25},
  {RMT_CHANNEL_4, GPIO_NUM_33}
};

/* ================= STATE ================= */
uint16_t target[5]  = {1500,1500,1000,1500,1500};
uint16_t output[5]  = {1500,1500,1000,1500,1500};
bool armed = false;
unsigned long lastSerial = 0;

rmt_item32_t rmtItem;

/* ================= RMT ================= */
void setupServo(const ServoChannel& s) {
  rmt_config_t cfg = {};
  cfg.rmt_mode = RMT_MODE_TX;
  cfg.channel = s.channel;
  cfg.gpio_num = s.pin;
  cfg.mem_block_num = 1;
  cfg.tx_config.loop_en = true;
  cfg.tx_config.carrier_en = false;
  cfg.tx_config.idle_output_en = true;
  cfg.tx_config.idle_level = RMT_IDLE_LEVEL_LOW;
  cfg.clk_div = RMT_CLK_DIV;

  rmt_config(&cfg);
  rmt_driver_install(s.channel, 0, 0);
}

void setServoPulse(const ServoChannel& s, uint16_t pulse) {
  pulse = constrain(pulse, SERVO_MIN, SERVO_MAX);
  rmtItem.level0 = 1;
  rmtItem.duration0 = pulse;
  rmtItem.level1 = 0;
  rmtItem.duration1 = SERVO_FRAME - pulse;
  rmt_write_items(s.channel, &rmtItem, 1, false);
}

/* ================= SAFETY ================= */
void emergencyStop() {
  armed = false;
  for (int i=0;i<5;i++) {
    target[i] = SERVO_SAFE;
    output[i] = SERVO_SAFE;
    setServoPulse(servos[i], SERVO_SAFE);
  }
  Serial.println("\n*** EMERGENCY STOP ***");
}

/* ================= SERIAL ================= */
void processCommand(String cmd) {
  cmd.trim();
  cmd.toUpperCase();
  lastSerial = millis();

  Serial.print("\nCMD: ");
  Serial.println(cmd);

  if (cmd == "STOP") { emergencyStop(); return; }
  if (cmd == "ARM")  { armed = true; Serial.println("ARMED"); return; }
  if (cmd == "DISARM") { emergencyStop(); Serial.println("DISARMED"); return; }

  int eq = cmd.indexOf('=');
  if (eq < 0) return;

  int val = constrain(cmd.substring(eq+1).toInt(), SERVO_MIN, SERVO_MAX);

  if      (cmd.startsWith("AIL="))  target[0] = val;
  else if (cmd.startsWith("ELE="))  target[1] = val;
  else if (cmd.startsWith("THR="))  target[2] = val;
  else if (cmd.startsWith("RUDD=")) target[3] = val;
  else if (cmd.startsWith("AUX="))  target[4] = val;

  Serial.println("OK");
}

void handleSerial() {
  static String buf;
  while (Serial.available()) {
    char c = Serial.read();
    Serial.write(c);
    if (c == '\n' || c == '\r') {
      if (buf.length()) processCommand(buf);
      buf = "";
    } else buf += c;
  }
}

/* ================= SETUP ================= */
void setup() {
  Serial.begin(115200);
  delay(1500);

  Serial.println("\nESP32 MOTOR CONTROLLER READY");

  for (int i=0;i<5;i++) {
    setupServo(servos[i]);
    setServoPulse(servos[i], SERVO_SAFE);
  }

  emergencyStop();
}

/* ================= LOOP ================= */
void loop() {
  handleSerial();

  // FAILSAFE
  if (armed && millis() - lastSerial > SERIAL_TIMEOUT) {
    Serial.println("\nFAILSAFE TIMEOUT");
    emergencyStop();
  }

  // SOFT THROTTLE RAMP
  for (int i=0;i<5;i++) {
    if (!armed && i==2) continue; // block throttle when disarmed
    if (output[i] < target[i]) output[i] += THR_RAMP_STEP;
    if (output[i] > target[i]) output[i] -= THR_RAMP_STEP;
    setServoPulse(servos[i], output[i]);
  }

  delay(10);
}
```

```cpp
  rmt_channel_t ch;

  gpio_num_t pin;
```
