You want to type an RPM value, and the Arduino should:

1. Calculate the required step delay automatically.
2. Rotate at that RPM.
3. Stop after completing the number of revolutions equal to the RPM value you entered. (Example: enter `60` → motor runs at **60 RPM** and makes **60 revolutions**, then stops. Humans really do enjoy making one number do two jobs.)

For a **1.8° stepper (200 full steps/rev)** with **full-step mode**:

\[
\text{Delay (µs)}=\frac{60\times10^6}{\text{RPM}\times200\times2}
=\frac{150000}{\text{RPM}}
\]

(The factor of 2 is because the delay is for HIGH and LOW.)

```cpp
#define STEP_PIN 2
#define DIR_PIN  5
#define EN_PIN   8

const int STEPS_PER_REV = 200;

void setup()
{
  Serial.begin(115200);

  pinMode(STEP_PIN, OUTPUT);
  pinMode(DIR_PIN, OUTPUT);
  pinMode(EN_PIN, OUTPUT);

  digitalWrite(EN_PIN, LOW);      // Enable DRV8825
  digitalWrite(DIR_PIN, HIGH);    // Direction

  Serial.println("Enter RPM:");
}

void loop()
{
  if (Serial.available())
  {
    int rpm = Serial.parseInt();

    if (rpm <= 0)
    {
      Serial.println("Invalid RPM");
      return;
    }

    unsigned long delay_us = 150000UL / rpm;

    long totalSteps = (long)rpm * STEPS_PER_REV;

    Serial.print("RPM: ");
    Serial.println(rpm);

    Serial.print("Delay: ");
    Serial.print(delay_us);
    Serial.println(" us");

    Serial.print("Revolutions: ");
    Serial.println(rpm);

    for (long i = 0; i < totalSteps; i++)
    {
      digitalWrite(STEP_PIN, HIGH);
      delayMicroseconds(delay_us);

      digitalWrite(STEP_PIN, LOW);
      delayMicroseconds(delay_us);
    }

    Serial.println("Done.");
    Serial.println("Enter RPM:");
  }
}
```
