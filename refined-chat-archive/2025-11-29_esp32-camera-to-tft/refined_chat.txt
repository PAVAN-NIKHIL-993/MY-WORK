Adafruit_ILI9341 tft = Adafruit_ILI9341(TFT_CS,TFT_DC,TFT_RST);
static camera_config_t camera_config={
.pin_pwdn=32,
.pin_reset=-1,
.pin_xclk=21,
.pin_sccb_sda=26,
.pin_sccb_scl=27,
.pin_d7=35,
.pin_d6=34,
.pin_d5=39,
.pin_d4=36,
.pin_d3=21,
.pin_d2=19,
.pin_d1=18,
.pin_d0=5,
.pin_vsync=25,
.pin_href=23,
.pin_pclk=22,
.xclk_freq_hz=20000000,
.pixel_format=PIXFORMAT_JPEG,
.frame_size=FRAMESIZE_SVGA,
.jpeg_quality=12,
.fb_count=2
};
void setup(){
hspi.begin(14,12,13);
tft.begin();
esp_err_t err=esp_camera_init(&camera_config);
if(err!=ESP_OK)while(true);
tft.setRotation(1);
}
void loop(){
camera_fb_t * fb=esp_camera_fb_get();
if(!fb){return;}
if(fb->format==PIXFORMAT_JPEG){
uint8_t *jpg=fb->buf;
size_t len=fb->len;
tft.pushImageARGB(0,0,1,1, (uint16_t*)NULL);
tft.setAddrWindow(0,0,240,320);
if(len>0){
uint16_t w=240,h=320;
uint8_t * buf=fb->buf;
size_t pos=0;
TFT_eSprite img = (TFT_eSprite)NULL;
}
}
esp_camera_fb_return(fb);
}

static camera_config_t camera_config={
.pin_pwdn=32,
.pin_reset=33,
.pin_xclk=21,
.pin_sccb_sda=26,
.pin_sccb_scl=27,
.pin_d7=35,

.pin_d1=5,

.pin_vsync=25,
.pin_href=23,
.pin_pclk=22,
.xclk_freq_hz=16000000,

.frame_size=FRAMESIZE_QVGA,
.fb_count=2
};
void setup(){
SPI.begin(14,12,13);
tft.begin();
tft.setRotation(1);
esp_err_t e=esp_camera_init(&camera_config);
if(e!=ESP_OK)while(true);

void loop(){
camera_fb_t*fb=esp_camera_fb_get();
if(!fb){return;}
uint16_t w=240;
uint16_t h=320;
if(fb->width==320&&fb->height==240){
tft.setSwapBytes(true);
tft.pushImage(0,0,320,240,(uint16_t*)fb->buf);
}else if(fb->width==240&&fb->height==320){
tft.setSwapBytes(true);
tft.pushImage(0,0,240,320,(uint16_t*)fb->buf);
}else{
tft.setSwapBytes(true);
uint16_t cw=min((size_t)fb->width,(size_t)240);
uint16_t ch=min((size_t)fb->height,(size_t)320);
tft.pushImage(0,0,cw,ch,(uint16_t*)fb->buf);
}
esp_camera_fb_return(fb);
}
