```cpp
  else if (cmd.startsWith("AUX="))  target[4] = value;
  else {
    Serial.println("ERR UNKNOWN");
    return;
  }

  Serial.println("OK");
}

void handleSerial() {
  static String buffer;

  while (Serial.available()) {
    char c = Serial.read();
    Serial.write(c);

    if (c == '\n' || c == '\r') {
      if (buffer.length()) processCommand(buffer);
      buffer = "";
    } else {
      buffer += c;
    }
  }
}

/* ================= SETUP ================= */
void setup() {
  Serial.begin(115200);
  delay(1000);

  Serial.println("\nESP32 RMT SERVO CONTROLLER READY");

  for (int i = 0; i < 5; i++) {
    setupServo(servos[i]);
    setServoPulse(servos[i], SERVO_SAFE);
  }

  emergencyStop();
}

/* ================= LOOP ================= */
void loop() {
  handleSerial();

  // FAILSAFE
  if (armed && millis() - lastSerial > SERIAL_TIMEOUT) {
    Serial.println("\nFAILSAFE TIMEOUT");
    emergencyStop();
  }

  // RAMP OUTPUTS
  for (int i = 0; i < 5; i++) {
    if (!armed && i == 2) continue; // block throttle when disarmed

    if (output[i] < target[i]) output[i] += THR_RAMP_STEP;
    if (output[i] > target[i]) output[i] -= THR_RAMP_STEP;

    setServoPulse(servos[i], output[i]);
  }

  delay(LOOP_DELAY);
}
```

## 🧠 Why this version doesn’t self-sabotage

- **No RMT loop_en abuse**  
  Looping RMT with changing pulse widths causes undefined behavior. We refresh every loop instead.

- **Correct 1 µs timing**  
  `clk_div = 80` means your pulse widths finally match reality.

- **ESC-safe arming logic**  
  Throttle is locked unless armed. Your motors stay attached to the table.

- **Failsafe that actually fires**  
  If serial dies, throttle drops. No runaway ESC drama.
