#include <Wire.h>

/* ===================== CONFIG ===================== */

#define I2C_SDA 8
#define I2C_SCL 9
#define I2C_FREQ 400000

#define MAX_SENSORS 2

uint8_t sensorAddresses[MAX_SENSORS] = {0x44, 0x45};

/* ===================== ENUMS ===================== */

enum Repeatability {
  REP_HIGH,
  REP_MED,
  REP_LOW
};

enum PeriodicRate {
  RATE_1HZ,
  RATE_2HZ,
  RATE_4HZ
};

/* ===================== SENSOR STRUCT ===================== */

struct SHT31 {
  uint8_t address;
  bool connected;

  uint32_t serialNumber;
  uint16_t status;

  float temperature;
  float humidity;

  uint16_t rawTemp;
  uint16_t rawHum;

  bool measurementPending;
  uint32_t measurementStart;

  bool periodicMode;
};

SHT31 sensors[MAX_SENSORS];

/* ===================== CRC8 ===================== */

uint8_t crc8(const uint8_t *data, int len) {
  uint8_t crc = 0xFF;
  for (int i = 0; i < len; i++) {
    crc ^= data[i];
    for (int b = 0; b < 8; b++) {
      crc = (crc & 0x80) ? (crc << 1) ^ 0x31 : (crc << 1);
    }
  }
  return crc;
}

/* ===================== LOW LEVEL ===================== */

bool writeCommand(uint8_t addr, uint16_t cmd) {
  Wire.beginTransmission(addr);
  Wire.write(cmd >> 8);
  Wire.write(cmd & 0xFF);
  return (Wire.endTransmission() == 0);
}

bool readBytes(uint8_t addr, uint8_t *buffer, uint8_t length) {
  Wire.requestFrom(addr, length);
  if (Wire.available() != length) return false;

  for (int i = 0; i < length; i++)
    buffer[i] = Wire.read();

  return true;
}

/* ===================== BASIC OPS ===================== */

bool isConnected(uint8_t addr) {
  Wire.beginTransmission(addr);
  return (Wire.endTransmission() == 0);
}

void softReset(uint8_t addr) {
  writeCommand(addr, 0x30A2);
  delay(10);
}

/* ===================== SERIAL NUMBER ===================== */

uint32_t getSerialNumber(uint8_t addr) {
  writeCommand(addr, 0x3780);
  delay(10);

  uint8_t data[6];
  if (!readBytes(addr, data, 6)) return 0;

  if (crc8(data, 2) != data[2]) return 0;
  if (crc8(data + 3, 2) != data[5]) return 0;

  return ((uint32_t)data[0] << 24) |
         ((uint32_t)data[1] << 16) |
         ((uint32_t)data[3] << 8)  |
          data[4];
}

/* ===================== STATUS ===================== */

uint16_t readStatus(uint8_t addr) {
  writeCommand(addr, 0xF32D);
  delay(5);

  uint8_t data[3];
  if (!readBytes(addr, data, 3)) return 0;

  if (crc8(data, 2) != data[2]) return 0;

  return (data[0] << 8) | data[1];
}

/* ===================== HEATER ===================== */

void heaterOn(uint8_t addr) {
  writeCommand(addr, 0x306D);
}

void heaterOff(uint8_t addr) {
  writeCommand(addr, 0x3066);
}

/* ===================== SINGLE SHOT ===================== */

void startMeasurement(SHT31 &sensor, Repeatability rep) {

  uint16_t cmd;

  if (rep == REP_HIGH) cmd = 0x2400;
  else if (rep == REP_MED) cmd = 0x240B;
  else cmd = 0x2416;

  writeCommand(sensor.address, cmd);

  sensor.measurementPending = true;
  sensor.measurementStart = millis();
}

bool readMeasurement(SHT31 &sensor) {

  if (!sensor.measurementPending) return false;
  if (millis() - sensor.measurementStart < 20) return false;

  uint8_t data[6];
  if (!readBytes(sensor.address, data, 6)) return false;

  if (crc8(data, 2) != data[2]) return false;
  if (crc8(data + 3, 2) != data[5]) return false;

  sensor.rawTemp = (data[0] << 8) | data[1];
  sensor.rawHum  = (data[3] << 8) | data[4];

  sensor.temperature = -45 + 175 * ((float)sensor.rawTemp / 65535.0);
  sensor.humidity    = 100 * ((float)sensor.rawHum / 65535.0);

  sensor.measurementPending = false;

  return true;
}

/* ===================== PERIODIC MODE ===================== */

void startPeriodic(SHT31 &sensor, Repeatability rep, PeriodicRate rate) {

  uint16_t cmd;

  if (rate == RATE_1HZ) {
    cmd = (rep == REP_HIGH) ? 0x2130 :
          (rep == REP_MED)  ? 0x2126 : 0x212D;
  }
  else if (rate == RATE_2HZ) {
    cmd = (rep == REP_HIGH) ? 0x2236 :
          (rep == REP_MED)  ? 0x2220 : 0x222B;
  }
  else { // 4Hz
    cmd = (rep == REP_HIGH) ? 0x2334 :
          (rep == REP_MED)  ? 0x2322 : 0x2329;
  }

  writeCommand(sensor.address, cmd);
  sensor.periodicMode = true;
}

void stopPeriodic(SHT31 &sensor) {
  writeCommand(sensor.address, 0x3093);
  sensor.periodicMode = false;
}

bool fetchPeriodic(SHT31 &sensor) {

  if (!sensor.periodicMode) return false;

  writeCommand(sensor.address, 0xE000);
  delay(5);

  uint8_t data[6];
  if (!readBytes(sensor.address, data, 6)) return false;

  if (crc8(data, 2) != data[2]) return false;
  if (crc8(data + 3, 2) != data[5]) return false;

  sensor.rawTemp = (data[0] << 8) | data[1];
  sensor.rawHum  = (data[3] << 8) | data[4];

  sensor.temperature = -45 + 175 * ((float)sensor.rawTemp / 65535.0);
  sensor.humidity    = 100 * ((float)sensor.rawHum / 65535.0);

  return true;
}

/* ===================== INIT ===================== */

void initSensor(SHT31 &sensor, uint8_t addr) {

  sensor.address = addr;
  sensor.connected = isConnected(addr);

  if (!sensor.connected) return;

  softReset(addr);

  sensor.serialNumber = getSerialNumber(addr);
  sensor.status = readStatus(addr);

  sensor.measurementPending = false;
  sensor.periodicMode = false;
}

/* ===================== SETUP ===================== */

void setup() {
  Serial.begin(115200);

  Wire.begin(I2C_SDA, I2C_SCL);
  Wire.setClock(I2C_FREQ);

  for (int i = 0; i < MAX_SENSORS; i++)
    initSensor(sensors[i], sensorAddresses[i]);
}

/* ===================== LOOP ===================== */

void loop() {

  for (int i = 0; i < MAX_SENSORS; i++) {

    if (!sensors[i].connected) continue;

    // Single-shot example
    if (!sensors[i].measurementPending) {
      startMeasurement(sensors[i], REP_HIGH);
    }

    if (readMeasurement(sensors[i])) {

      Serial.print("Sensor ");
      Serial.print(i);
      Serial.print(" | Temp: ");
      Serial.print(sensors[i].temperature);
      Serial.print(" C | Hum: ");
      Serial.print(sensors[i].humidity);
      Serial.println(" %");
    }
  }

  delay(100);
}

# 🔥 What “Heater On” Actually Is

The **SHT31 has a tiny internal resistive heater** built into the chip.

When you send:

```cpp
writeCommand(addr, 0x306D);  // Heater ON
```

You are telling the sensor to **intentionally warm itself up**.

And yes, that sounds counterproductive for a temperature sensor.

---

# 🤨 Why Would Anyone Heat a Temperature Sensor?

Because humidity sensors hate condensation.

In environments like:
- Near 100% humidity  
- Outdoor weather stations  
- Salt-heavy air  
- Cold-to-warm transitions  

Water can condense on the sensing element.

When that happens:
- Humidity reads 100%
- Temperature readings drift
- Sensor response slows down

The heater evaporates moisture from the sensing membrane.

It’s a **self-drying function**, not a room heater.

---

# 📈 What Happens When Heater Is ON

- Sensor temperature reading increases artificially  
- Humidity reading drops artificially  
- Power consumption increases  
- Chip gets warm  

So you **never leave it on permanently**.

It’s meant for short bursts.

---

# 🧠 When Should You Use It?

Use heater:
- If RH > 90% for long periods
- If sensor is outdoors
- If you detect saturation
- After condensation events

Do NOT use heater:
- For normal indoor monitoring
- For precise environmental logging
- For HVAC control without compensation
