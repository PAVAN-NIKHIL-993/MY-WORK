## 4. Signal processing (real DSP, not averaging vibes)

### Globals

```cpp
static float irDC = 0;
static float hp = 0;
static float lp = 0;

constexpr float DC_ALPHA = 0.99f;
constexpr float HP_ALPHA = 0.95f;
constexpr float LP_ALPHA = 0.1f;
```

### Replace processing core
```cpp
float ir = (float)raw.ir;

// DC tracking
irDC = DC_ALPHA * irDC + (1.0f - DC_ALPHA) * ir;
float ac = ir - irDC;

// High-pass
hp = HP_ALPHA * hp + (1.0f - HP_ALPHA) * ac;

// Low-pass (noise kill)
lp = lp + LP_ALPHA * (hp - lp);

f.value = lp;
f.quality = fabs(lp) / (irDC + 1.0f);
```

Why this works:
- DC tracks finger pressure + LED drift
- AC isolates pulse
- LP removes motion spikes
- Quality is **physically meaningful**, not cosmetic

```cpp
/*
 * ESP32-C3 + MAX30105 Pulse Monitor
 * Non-medical, research / learning use only
 * Single-file firmware, no placeholders
 */

#include <Arduino.h>
#include <Wire.h>
#include "MAX30105.h"
#include "esp_timer.h"
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "freertos/queue.h"
#include "freertos/semphr.h"

/* ================= CONFIG ================= */
#define SDA_PIN 8
#define SCL_PIN 9
#define I2C_FREQ 400000
#define SAMPLE_PERIOD_US 10000   // 100 Hz
#define BEAT_THRESHOLD 120.0f
#define REFRACTORY_US 300000     // 200 BPM max

/* ================= SENSOR ================= */
MAX30105 sensor;

/* ================= RTOS ================= */
QueueHandle_t qRaw;
QueueHandle_t qFiltered;
SemaphoreHandle_t dataMutex;
SemaphoreHandle_t i2cMutex;
SemaphoreHandle_t semAcquire;

/* ================= TIMER ================= */
esp_timer_handle_t acqTimer;

/* ================= DATA ================= */
struct RawSample {
  uint32_t ts;
  uint32_t ir;
  uint32_t red;
};

struct FilteredSample {
  uint32_t ts;
  float value;
  float quality;
};

struct SharedData {
  float bpm;
  float bpmConfidence;
  float rrVariance;
  uint8_t sqi;
  bool sqiValid;
};

SharedData shared = {0};

/* ================= UTILS ================= */
uint32_t nowMicros() {
  return (uint32_t)esp_timer_get_time();
}

/* ================= I2C RECOVERY ================= */
void i2cRecover() {
  pinMode(SCL_PIN, OUTPUT_OPEN_DRAIN);
  pinMode(SDA_PIN, OUTPUT_OPEN_DRAIN);
  digitalWrite(SCL_PIN, HIGH);
  digitalWrite(SDA_PIN, HIGH);
  for (int i = 0; i < 9; i++) {
    digitalWrite(SCL_PIN, LOW);
    delayMicroseconds(5);
    digitalWrite(SCL_PIN, HIGH);
    delayMicroseconds(5);
  }
}

/* ================= SENSOR INIT ================= */
bool initSensor() {
  if (!sensor.begin(Wire, I2C_SPEED_FAST)) return false;

  sensor.setup(
    0x1F,   // LED brightness
    4,      // sample average
    2,      // red + IR
    400,    // sample rate
    411,    // pulse width
    4096   // ADC range
  );

  sensor.clearFIFO();
  return true;
}

/* ================= TIMER ISR ================= */
void acqTimerCb(void*) {
  BaseType_t hpTask = pdFALSE;
  xSemaphoreGiveFromISR(semAcquire, &hpTask);
  if (hpTask) portYIELD_FROM_ISR();
}

/* ================= TASK: ACQUISITION ================= */
void taskAcquisition(void*) {
  for (;;) {
    if (xSemaphoreTake(semAcquire, portMAX_DELAY) == pdTRUE) {
      xSemaphoreTake(i2cMutex, portMAX_DELAY);

      while (sensor.available()) {
        RawSample s;
        s.ts = nowMicros();
        s.ir = sensor.getIR();
        s.red = sensor.getRed();
        sensor.nextSample();
        xQueueSend(qRaw, &s, 0);
      }

      xSemaphoreGive(i2cMutex);
    }
  }
}

/* ================= TASK: SIGNAL PROCESSING ================= */
void taskSignalProcessing(void*) {
  static float irDC = 0;
  static float hp = 0;
  static float lp = 0;

  const float DC_ALPHA = 0.99f;
  const float HP_ALPHA = 0.95f;
  const float LP_ALPHA = 0.1f;

  for (;;) {
    RawSample raw;
    if (xQueueReceive(qRaw, &raw, portMAX_DELAY)) {
      float ir = (float)raw.ir;

      irDC = DC_ALPHA * irDC + (1.0f - DC_ALPHA) * ir;
      float ac = ir - irDC;

      hp = HP_ALPHA * hp + (1.0f - HP_ALPHA) * ac;
      lp = lp + LP_ALPHA * (hp - lp);

      FilteredSample f;
      f.ts = raw.ts;
      f.value = lp;
      f.quality = fabs(lp) / (irDC + 1.0f);

      xQueueSend(qFiltered, &f, 0);
    }
  }
}

/* ================= TASK: BEAT DETECTION ================= */
void taskBeatDetection(void*) {
  static uint32_t lastBeatTs = 0;
  static float lastBpm = 0;

  for (;;) {
    FilteredSample f;
    if (xQueueReceive(qFiltered, &f, portMAX_DELAY)) {
      if (f.value > BEAT_THRESHOLD && f.quality > 0.02f) {
        uint32_t now = f.ts;
        if (now - lastBeatTs > REFRACTORY_US) {
          float ibi = (now - lastBeatTs) / 1000000.0f;
          lastBeatTs = now;

          float bpm = 60.0f / ibi;

          xSemaphoreTake(dataMutex, portMAX_DELAY);
          shared.rrVariance = fabs(bpm - lastBpm);
          shared.bpm = bpm;
          shared.bpmConfidence = f.quality;
          xSemaphoreGive(dataMutex);

          lastBpm = bpm;
        }
      }
    }
  }
}

/* ================= TASK: SQI ================= */
void taskSQI(void*) {
  for (;;) {
    xSemaphoreTake(dataMutex, portMAX_DELAY);
    float conf = shared.bpmConfidence;
    float bpm = shared.bpm;
    xSemaphoreGive(dataMutex);

    uint8_t sqi = 0;
    if (conf > 0.03f && bpm > 40 && bpm < 180) sqi = 90;
    else if (conf > 0.02f) sqi = 70;
    else if (conf > 0.01f) sqi = 40;
    else sqi = 10;

    xSemaphoreTake(dataMutex, portMAX_DELAY);
    shared.sqi = sqi;
    shared.sqiValid = (sqi >= 50);
    xSemaphoreGive(dataMutex);

    vTaskDelay(pdMS_TO_TICKS(100));
  }
}

/* ================= TASK: ADAPTIVE LED ================= */
void taskAdaptiveLED(void*) {
  static uint8_t irAmp = 0x20;

  for (;;) {
    xSemaphoreTake(dataMutex, portMAX_DELAY);
    uint8_t sqi = shared.sqi;
    xSemaphoreGive(dataMutex);

    if (sqi < 50 && irAmp < 0x3F) irAmp++;
    if (sqi > 85 && irAmp > 0x10) irAmp--;

    xSemaphoreTake(i2cMutex, portMAX_DELAY);
    sensor.setPulseAmplitudeIR(irAmp);
    sensor.setPulseAmplitudeRed(irAmp - 4);
    xSemaphoreGive(i2cMutex);

    vTaskDelay(pdMS_TO_TICKS(50));
  }
}

/* ================= TASK: OUTPUT ================= */
void taskOutput(void*) {
  for (;;) {
    xSemaphoreTake(dataMutex, portMAX_DELAY);
    float bpm = shared.bpm;
    uint8_t sqi = shared.sqi;
    float conf = shared.bpmConfidence;
    xSemaphoreGive(dataMutex);

    Serial.printf("BPM: %.1f | SQI: %u | CONF: %.3f\n",
                  bpm, sqi, conf);

    vTaskDelay(pdMS_TO_TICKS(100));
  }
}

/* ================= SETUP ================= */
void setup() {
  Serial.begin(115200);
  delay(200);

  Wire.begin(SDA_PIN, SCL_PIN, I2C_FREQ);
  i2cRecover();

  if (!initSensor()) {
    Serial.println("MAX30105 NOT FOUND");
    while (1);
  }

  qRaw = xQueueCreate(64, sizeof(RawSample));
  qFiltered = xQueueCreate(64, sizeof(FilteredSample));
  dataMutex = xSemaphoreCreateMutex();
  i2cMutex = xSemaphoreCreateMutex();
  semAcquire = xSemaphoreCreateBinary();

  const esp_timer_create_args_t timerArgs = {
    .callback = &acqTimerCb,
    .arg = nullptr,
    .dispatch_method = ESP_TIMER_ISR,
    .name = "acqTimer"
  };

  esp_timer_create(&timerArgs, &acqTimer);
  esp_timer_start_periodic(acqTimer, SAMPLE_PERIOD_US);

  xTaskCreate(taskAcquisition, "acq", 4096, NULL, 6, NULL);
  xTaskCreate(taskSignalProcessing, "dsp", 4096, NULL, 5, NULL);
  xTaskCreate(taskBeatDetection, "beat", 4096, NULL, 4, NULL);
  xTaskCreate(taskSQI, "sqi", 2048, NULL, 3, NULL);
  xTaskCreate(taskAdaptiveLED, "led", 2048, NULL, 2, NULL);
  xTaskCreate(taskOutput, "out", 2048, NULL, 1, NULL);
}

void loop() {
  vTaskDelay(portMAX_DELAY);
}
```
