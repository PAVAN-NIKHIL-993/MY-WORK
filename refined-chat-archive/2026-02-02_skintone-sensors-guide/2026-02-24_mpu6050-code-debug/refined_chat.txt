// ==================== KALMAN FILTER ====================
struct KalmanFilter {
  float Q_angle;
  float Q_bias;
  float R_measure;
  float angle;
  float bias;
  float rate;
  float P[2][2];
};

KalmanFilter kalman_roll, kalman_pitch;

void initKalman(KalmanFilter *k) {
  k->Q_angle = 0.001;
  k->Q_bias = 0.003;
  k->R_measure = 0.03;
  k->angle = 0.0;
  k->bias = 0.0;
  k->P[0][0] = 0.0;
  k->P[0][1] = 0.0;
  k->P[1][0] = 0.0;
  k->P[1][1] = 0.0;
}

float kalmanUpdate(KalmanFilter *k, float newAngle, float newRate, float dt) {
  // Predict
  k->rate = newRate - k->bias;
  k->angle += dt * k->rate;
  
  k->P[0][0] += dt * (dt * k->P[1][1] - k->P[0][1] - k->P[1][0] + k->Q_angle);
  k->P[0][1] -= dt * k->P[1][1];
  k->P[1][0] -= dt * k->P[1][1];
  k->P[1][1] += k->Q_bias * dt;
  
  // Update
  float S = k->P[0][0] + k->R_measure;
  float K[2];
  K[0] = k->P[0][0] / S;
  K[1] = k->P[1][0] / S;
  
  float y = newAngle - k->angle;
  k->angle += K[0] * y;
  k->bias += K[1] * y;
  
  float P00_temp = k->P[0][0];
  float P01_temp = k->P[0][1];
  
  k->P[0][0] -= K[0] * P00_temp;
  k->P[0][1] -= K[0] * P01_temp;
  k->P[1][0] -= K[1] * P00_temp;
  k->P[1][1] -= K[1] * P01_temp;
  
  return k->angle;
}

  Wire.endTransmission(false);
  Wire.requestFrom(MPU6050_ADDR, 14, true);

  s.ax_raw = Wire.read()<<8 | Wire.read();
  s.ay_raw = Wire.read()<<8 | Wire.read();
  s.az_raw = Wire.read()<<8 | Wire.read();
  Wire.read(); Wire.read(); // skip temp
  s.gx_raw = Wire.read()<<8 | Wire.read();
  s.gy_raw = Wire.read()<<8 | Wire.read();
  s.gz_raw = Wire.read()<<8 | Wire.read();
}

// ================== MOVING AVERAGE ==================
float movingAvg(float val, float *buf, uint8_t &idx) {
  buf[idx] = val;
  idx = (idx + 1) % MOVING_AVG_SAMPLES;

  float sum = 0;
  for(int i=0;i<MOVING_AVG_SAMPLES;i++) sum += buf[i];
  return sum / MOVING_AVG_SAMPLES;
}

// ================== PROCESS ==================
void processIMU() {

  s.ax = (float)s.ax_raw / ACCEL_SENS;
  s.ay = (float)s.ay_raw / ACCEL_SENS;
  s.az = (float)s.az_raw / ACCEL_SENS;

  s.gx = (float)s.gx_raw / GYRO_SENS;
  s.gy = (float)s.gy_raw / GYRO_SENS;
  s.gz = (float)s.gz_raw / GYRO_SENS;

  s.ax = movingAvg(s.ax, s.ax_buf, s.ax_i);
  s.ay = movingAvg(s.ay, s.ay_buf, s.ay_i);
  s.az = movingAvg(s.az, s.az_buf, s.az_i);

  s.gx = movingAvg(s.gx, s.gx_buf, s.gx_i);
  s.gy = movingAvg(s.gy, s.gy_buf, s.gy_i);
  s.gz = movingAvg(s.gz, s.gz_buf, s.gz_i);
}

// ================== ANGLES ==================
void computeAngles() {

  unsigned long now = micros();
  s.dt = (now - s.lastTime) / 1000000.0;
  s.lastTime = now;

  float accRoll  = atan2(s.ay, s.az) * RAD_TO_DEG;
  float accPitch = atan2(-s.ax, sqrt(s.ay*s.ay + s.az*s.az)) * RAD_TO_DEG;

  s.roll  = kalmanUpdate(kRoll, accRoll, s.gx, s.dt);
  s.pitch = kalmanUpdate(kPitch, accPitch, s.gy, s.dt);

  s.yaw += s.gz * s.dt;
}

// ================== STREAM ==================
void streamData() {
  Serial.print(s.roll,4); Serial.print(",");
  Serial.print(s.pitch,4); Serial.print(",");
  Serial.print(s.yaw,4); Serial.print(",");
  Serial.print(s.ax,4); Serial.print(",");
  Serial.print(s.ay,4); Serial.print(",");
  Serial.print(s.az,4); Serial.print(",");
  Serial.print(s.gx,4); Serial.print(",");
  Serial.print(s.gy,4); Serial.print(",");
  Serial.println(s.gz,4);
}

// ================== SETUP ==================
void setup() {
  Serial.begin(115200);
  Wire.begin();
  Wire.setClock(400000);

  if (Wire.requestFrom(MPU6050_ADDR, 1) == 0) {
    while(1);
  }

  writeReg(PWR_MGMT_1, 0x00);
  writeReg(SMPLRT_DIV, 0);
  writeReg(CONFIG, 0x01);
  writeReg(GYRO_CONFIG, 0x00);
  writeReg(ACCEL_CONFIG, 0x00);

  memset(&s, 0, sizeof(s));
  s.lastTime = micros();
}

// ================== LOOP ==================
void loop() {

  readRaw();
  processIMU();
  computeAngles();

  unsigned long now = millis();

  if (now - lastStreamTime >= streamInterval) {
    lastStreamTime = now;
    streamData();
  }
}
