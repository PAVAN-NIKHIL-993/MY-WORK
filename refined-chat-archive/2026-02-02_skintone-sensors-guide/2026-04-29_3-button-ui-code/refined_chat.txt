### 📁 `main.ino`

```cpp
#include <WiFi.h>
#include <WebServer.h>

// WiFi Credentials
const char* ssid = "YOUR_WIFI";
const char* password = "YOUR_PASS";

// Pins
#define BTN1 4
#define BTN2 5
#define BTN3 6

#define LED1 10
#define LED2 11
#define LED3 12

#define MOTOR 9

WebServer server(80);

// ===== FUNCTIONS =====
void allOff() {
  digitalWrite(LED1, LOW);
  digitalWrite(LED2, LOW);
  digitalWrite(LED3, LOW);
  digitalWrite(MOTOR, LOW);
}

void vibrate(int duration) {
  digitalWrite(MOTOR, HIGH);
  delay(duration);
  digitalWrite(MOTOR, LOW);
}

void action1() {
  allOff();
  digitalWrite(LED1, HIGH);
  vibrate(200);
}

void action2() {
  allOff();
  digitalWrite(LED2, HIGH);
  vibrate(500);
}

void action3() {
  allOff();
  digitalWrite(LED3, HIGH);
  vibrate(1000);
}

// ===== WEB UI =====
String webpage = R"rawliteral(
<!DOCTYPE html>
<html>
<head>
  <title>Control Panel</title>
  <style>
    body { text-align:center; font-family:Arial; }
    button {
      padding:20px;
      margin:10px;
      font-size:20px;
      width:200px;
    }
  </style>
</head>
<body>
  <h2>3 Button Control</h2>

  <button onclick="send('b1')">Button 1</button><br>
  <button onclick="send('b2')">Button 2</button><br>
  <button onclick="send('b3')">Button 3</button>

<script>
function send(cmd){
  fetch("/" + cmd);
}
</script>

</body>
</html>
)rawliteral";

// ===== ROUTES =====
void handleRoot() {
  server.send(200, "text/html", webpage);
}

void handleB1() { action1(); server.send(200, "text/plain", "OK"); }
void handleB2() { action2(); server.send(200, "text/plain", "OK"); }
void handleB3() { action3(); server.send(200, "text/plain", "OK"); }

// ===== SETUP =====
void setup() {
  Serial.begin(115200);

  pinMode(BTN1, INPUT_PULLUP);
  pinMode(BTN2, INPUT_PULLUP);
  pinMode(BTN3, INPUT_PULLUP);

  pinMode(LED1, OUTPUT);
  pinMode(LED2, OUTPUT);
  pinMode(LED3, OUTPUT);

  pinMode(MOTOR, OUTPUT);

  allOff();

  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }

  Serial.println("\nConnected!");
  Serial.println(WiFi.localIP());

  server.on("/", handleRoot);
  server.on("/b1", handleB1);
  server.on("/b2", handleB2);
  server.on("/b3", handleB3);

  server.begin();
}

// ===== LOOP =====
void loop() {
  server.handleClient();

  // Physical Buttons
  if (!digitalRead(BTN1)) {
    action1();
    delay(300);
  }
  if (!digitalRead(BTN2)) {
    action2();
    delay(300);
  }
  if (!digitalRead(BTN3)) {
    action3();
    delay(300);
  }
}
```

## ⚠️ Critical Reality Check
- Motor **must** use transistor + diode. Direct pin = goodbye ESP32.
- GPIO 6–11 are flash pins on many ESP32 boards. If yours freaks out, change pins. Don’t blame physics.
- Buttons use `INPUT_PULLUP`, so logic is inverted (pressed = LOW). Yes, confusing. Welcome to embedded.

#include <WiFi.h>
#include <WebServer.h>

// WiFi Credentials
const char* ssid = "YOUR_WIFI";
const char* password = "YOUR_PASS";

// Pins
#define BTN1 4
#define BTN2 5
#define BTN3 6

#define LED1 10
#define LED2 11
#define LED3 12

#define MOTOR 9

WebServer server(80);

// ===== FUNCTIONS =====
void allOff() {
  digitalWrite(LED1, LOW);
  digitalWrite(LED2, LOW);
  digitalWrite(LED3, LOW);
  digitalWrite(MOTOR, LOW);
  Serial.println("[SYSTEM] All outputs OFF");
}

void vibrate(int duration) {
  Serial.print("[MOTOR] Vibrating for ");
  Serial.print(duration);
  Serial.println(" ms");

  digitalWrite(MOTOR, HIGH);
  delay(duration);
  digitalWrite(MOTOR, LOW);

  Serial.println("[MOTOR] Stopped");
}

void action1() {
  Serial.println("[ACTION] Button 1 Triggered → LED1 + Short Vibrate");
  allOff();
  digitalWrite(LED1, HIGH);
  vibrate(200);
}

void action2() {
  Serial.println("[ACTION] Button 2 Triggered → LED2 + Medium Vibrate");
  allOff();
  digitalWrite(LED2, HIGH);
  vibrate(500);
}

void action3() {
  Serial.println("[ACTION] Button 3 Triggered → LED3 + Long Vibrate");
  allOff();
  digitalWrite(LED3, HIGH);
  vibrate(1000);
}

// ===== WEB UI =====
String webpage = R"rawliteral(
<!DOCTYPE html>
<html>
<head>
  <title>Control Panel</title>
  <style>
    body { text-align:center; font-family:Arial; }
    button {
      padding:20px;
      margin:10px;
      font-size:20px;
      width:200px;
    }
  </style>
</head>
<body>
  <h2>3 Button Control</h2>

  <button onclick="send('b1')">Button 1</button><br>
  <button onclick="send('b2')">Button 2</button><br>
  <button onclick="send('b3')">Button 3</button>

<script>
function send(cmd){
  fetch("/" + cmd);
}
</script>

</body>
</html>
)rawliteral";

// ===== ROUTES =====
void handleRoot() {
  Serial.println("[WEB] UI Loaded");
  server.send(200, "text/html", webpage);
}

void handleB1() {
  Serial.println("[WEB] Command: Button 1");
  action1();
  server.send(200, "text/plain", "OK");
}

void handleB2() {
  Serial.println("[WEB] Command: Button 2");
  action2();
  server.send(200, "text/plain", "OK");
}

void handleB3() {
  Serial.println("[WEB] Command: Button 3");
  action3();
  server.send(200, "text/plain", "OK");
}

// ===== SETUP =====
void setup() {
  Serial.begin(115200);
  Serial.println("\n=== SYSTEM BOOT ===");

  pinMode(BTN1, INPUT_PULLUP);
  pinMode(BTN2, INPUT_PULLUP);
  pinMode(BTN3, INPUT_PULLUP);

  pinMode(LED1, OUTPUT);
  pinMode(LED2, OUTPUT);
  pinMode(LED3, OUTPUT);

  pinMode(MOTOR, OUTPUT);

  allOff();

  Serial.println("[WIFI] Connecting...");
  WiFi.begin(ssid, password);

  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }

  Serial.println("\n[WIFI] Connected!");
  Serial.print("[WIFI] IP Address: ");
  Serial.println(WiFi.localIP());

  server.on("/", handleRoot);
  server.on("/b1", handleB1);
  server.on("/b2", handleB2);
  server.on("/b3", handleB3);

  server.begin();
  Serial.println("[SERVER] HTTP Server Started");
}

// ===== LOOP =====
void loop() {
  server.handleClient();

  if (!digitalRead(BTN1)) {
    Serial.println("[BUTTON] Physical Button 1 Pressed");
    action1();
    delay(300);
  }

  if (!digitalRead(BTN2)) {
    Serial.println("[BUTTON] Physical Button 2 Pressed");
    action2();
    delay(300);
  }

  if (!digitalRead(BTN3)) {
    Serial.println("[BUTTON] Physical Button 3 Pressed");
    action3();
    delay(300);
  }
}

#include <WiFi.h>
#include <WebServer.h>

// Pins
#define BTN1 13

#define LED1 26

#define LED3 33

#define MOTOR 32

WebServer server(80);

// ===== STATE =====
int activeRequest = 0;
bool acknowledged = false;

unsigned long startTime = 0;
unsigned long lastMotorToggle = 0;
bool motorState = false;

// ===== LOG SYSTEM =====
String logs = "";

void addLog(String msg) {
  logs += msg + "<br>";
  if (logs.length() > 3000) logs = ""; // prevent overflow meltdown
}

// ===== CORE FUNCTIONS =====
void allOff() {
  digitalWrite(LED1, LOW);
  digitalWrite(LED2, LOW);
  digitalWrite(LED3, LOW);
  digitalWrite(MOTOR, LOW);

  activeRequest = 0;
  acknowledged = false;

  addLog("[SYSTEM] All OFF");
}

void acknowledge() {
  acknowledged = true;
  digitalWrite(MOTOR, LOW);

  digitalWrite(LED1, LOW);
  digitalWrite(LED2, LOW);
  digitalWrite(LED3, LOW);

  activeRequest = 0;

  addLog("[ACK] Request completed → System reset");
}

void startRequest(int req) {
  allOff();

  activeRequest = req;
  acknowledged = false;
  startTime = millis();

  addLog("[WEB] Button " + String(req) + " triggered");

  if (req == 1) digitalWrite(LED1, HIGH);
  if (req == 2) digitalWrite(LED2, HIGH);
  if (req == 3) digitalWrite(LED3, HIGH);

  addLog("[LED] LED " + String(req) + " ON");
}

// ===== MOTOR CONTROL =====
void handleMotor() {
  if (activeRequest == 0 || acknowledged) return;

  if (millis() - lastMotorToggle > 300) {
    motorState = !motorState;
    digitalWrite(MOTOR, motorState);
    lastMotorToggle = millis();
  }
}

// ===== TIMEOUT =====
void checkTimeout() {
  if (activeRequest != 0 && !acknowledged) {
    if (millis() - startTime > 30000) {
      addLog("[TIMEOUT] No response → shutting down");
      allOff();
    }
  }
}

// ===== BUTTON CHECK =====
void checkButtons() {
  if (activeRequest == 0 || acknowledged) return;

  if (activeRequest == 1 && !digitalRead(BTN1)) {
    addLog("[INPUT] Button 1 pressed");
    acknowledge();
  }

  if (activeRequest == 2 && !digitalRead(BTN2)) {
    addLog("[INPUT] Button 2 pressed");
    acknowledge();
  }

  if (activeRequest == 3 && !digitalRead(BTN3)) {
    addLog("[INPUT] Button 3 pressed");
    acknowledge();
  }
}

// ===== ROUTES =====
void handleRoot() {
  server.send(200, "text/html", webpage);
}

void handleLogs() {
  server.send(200, "text/html", logs);
}

void handleB1() { startRequest(1); server.send(200, "text/plain", "OK"); }
void handleB2() { startRequest(2); server.send(200, "text/plain", "OK"); }
void handleB3() { startRequest(3); server.send(200, "text/plain", "OK"); }

// ===== SETUP =====
void setup() {
  pinMode(BTN1, INPUT_PULLUP);
  pinMode(BTN2, INPUT_PULLUP);
  pinMode(BTN3, INPUT_PULLUP);

  pinMode(LED1, OUTPUT);
  pinMode(LED2, OUTPUT);
  pinMode(LED3, OUTPUT);

  pinMode(MOTOR, OUTPUT);

  allOff();

  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) delay(500);

  server.on("/", handleRoot);
  server.on("/logs", handleLogs);
  server.on("/b1", handleB1);
  server.on("/b2", handleB2);
  server.on("/b3", handleB3);

  server.begin();
}

// ===== LOOP =====
void loop() {
  server.handleClient();
  handleMotor();
  checkButtons();
  checkTimeout();
}
