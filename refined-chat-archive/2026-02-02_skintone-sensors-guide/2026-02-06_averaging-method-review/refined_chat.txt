## Arduino-style implementation (drop-in)

```cpp
#define TOLERANCE 3
#define HOLD_TIME 6000  // ms

int refValue = 0;
unsigned long refStartTime = 0;
bool firstRun = true;

int processValue(int currentValue) {
  unsigned long now = millis();

  // First sample initializes everything
  if (firstRun) {
    refValue = currentValue;
    refStartTime = now;
    firstRun = false;
    return refValue;
  }

  // If value jumps outside ±3 → reset immediately
  if (abs(currentValue - refValue) > TOLERANCE) {
    refValue = currentValue;
    refStartTime = now;
    return refValue;
  }

  // If stable for 6 seconds → accept latest value
  if (now - refStartTime >= HOLD_TIME) {
    refValue = currentValue;
    refStartTime = now;
    return refValue;
  }

  // Otherwise keep showing locked value
  return refValue;
}
```

Usage inside `loop()`:

```cpp
int raw = readSensor();      // whatever your source is

Serial.println(stable);
```

```cpp
  if (ir < IR_CONTACT_THRESHOLD) {
    if (!noContactState) {
      Serial.println("NO CONTACT");
      resetCalibration();
      noContactState = true;
    }
    delay(200);
    return;
  }
  noContactState = false;

  /* -------- FILTERING -------- */
  irFiltered = alpha * irFiltered + (1 - alpha) * ir;

  irBandpass[2] = irBandpass[1];
  irBandpass[1] = irBandpass[0];
  irBandpass[0] = irFiltered;

  irProcessed =
    irBandpass[0] * 0.30 +
    irBandpass[1] * 0.50 +
    irBandpass[2] * 0.20;

  /* -------- CALIBRATION -------- */
  if (isCalibrating) {
    calibrationSamples++;
    peakThreshold = max(peakThreshold, irProcessed);
    valleyThreshold = min(valleyThreshold, irProcessed);

    if (calibrationSamples > 200) {
      isCalibrating = false;
      Serial.println("READY");
    }
    return;
  }

  /* -------- ADAPTIVE THRESHOLD -------- */
  peakThreshold = 0.998 * peakThreshold + 0.002 * irProcessed;
  valleyThreshold = 0.998 * valleyThreshold + 0.002 * irProcessed;

  float dynamicThreshold =
    valleyThreshold + (peakThreshold - valleyThreshold) * 0.6;

  /* -------- MOTION -------- */
  float variance = abs(irProcessed - irPrev) + abs(irPrev - irPrev2);
  signalVariance = 0.9 * signalVariance + 0.1 * variance;

  bool motionDetected =
    signalVariance > (peakThreshold - valleyThreshold) * 0.4;

  /* -------- DERIVATIVES -------- */
  float firstDerivative = irProcessed - irPrev;
  float secondDerivative = firstDerivative - (irPrev - irPrev2);

  irPrev2 = irPrev;
  irPrev = irProcessed;

  if (irProcessed > dynamicThreshold)
    samplesAboveThreshold++;
  else
    samplesAboveThreshold = 0;

  /* -------- HEART BEAT -------- */
  if (firstDerivative > 0 &&
      !risingEdge &&
      samplesAboveThreshold > 3 &&
      secondDerivative < 0 &&
      !motionDetected &&
      millis() - lastBeatTime > 300) {

    unsigned long now = millis();
    unsigned long interval = now - lastBeatTime;

    if (interval > 300 && interval < 1500 && lastBeatTime > 0) {
      float instantBPM = 60000.0 / interval;

      if (bpmAvg == 0 || abs(instantBPM - bpmAvg) < bpmAvg * 0.25) {
        bpmHistory[bpmIndex] = instantBPM;
        bpmIndex = (bpmIndex + 1) % BPM_HISTORY;
        bpmAvg = calculateWeightedMedianBPM();
      }
    }

    lastBeatTime = now;
    risingEdge = true;
  }

  if (firstDerivative < 0) risingEdge = false;

  /* -------- PRINT -------- */
  if (millis() - lastPrint > 1000) {
    lastPrint = millis();

    if (bpmAvg > 0) {
      float stable = stabilizeBPM(bpmAvg);
      Serial.print("HR: ");
      Serial.print(stable, 1);
      Serial.println(" BPM");
    } else {
      Serial.println("Detecting...");
    }
  }
}

/* ================= HELPERS ================= */
void resetCalibration() {
  isCalibrating = true;
  calibrationSamples = 0;
  peakThreshold = 0;
  valleyThreshold = 1e9;
  bpmAvg = 0;

  bpmLocked = 0;
  bpmLockInit = false;
  bpmLockStart = 0;

  for (int i = 0; i < BPM_HISTORY; i++) bpmHistory[i] = 0;
}

float calculateWeightedMedianBPM() {
  float sorted[BPM_HISTORY];
  int count = 0;

  for (int i = 0; i < BPM_HISTORY; i++)
    if (bpmHistory[i] > 0)
      sorted[count++] = bpmHistory[i];

  if (count == 0) return 0;
  if (count == 1) return sorted[0];

  for (int i = 0; i < count - 1; i++)
    for (int j = 0; j < count - i - 1; j++)
      if (sorted[j] > sorted[j + 1]) {
        float t = sorted[j];
        sorted[j] = sorted[j + 1];
        sorted[j + 1] = t;
      }

  int start = count / 5;
  int end = count - start;
  float sum = 0;

  for (int i = start; i < end; i++) sum += sorted[i];
  return sum / (end - start);
}

float stabilizeBPM(float currentBPM) {
  unsigned long now = millis();

  if (!bpmLockInit && currentBPM > 0) {
    bpmLocked = currentBPM;
    bpmLockStart = now;
    bpmLockInit = true;
    return bpmLocked;
  }

  if (!bpmLockInit) return 0;

  if (abs(currentBPM - bpmLocked) > BPM_TOLERANCE) {
    bpmLocked = currentBPM;
    bpmLockStart = now;
    return bpmLocked;
  }

  if (now - bpmLockStart >= BPM_HOLD_TIME) {
    bpmLocked = currentBPM;
    bpmLockStart = now;
  }

  return bpmLocked;
}
```
