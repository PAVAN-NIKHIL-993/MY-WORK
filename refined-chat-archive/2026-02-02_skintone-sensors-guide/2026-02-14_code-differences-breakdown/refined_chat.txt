### Suggestions & Next Steps

- Power the ESP32 from a clean 3.3 V rail: ≥100 µF bulk + 100 nF ceramic at the VDD pins, and a separate regulator for servos/motors — brownouts masquerade as random firmware bugs.
- Respect boot strapping pins (GPIO 0/2/12 on classic ESP32, GPIO 2/8/9 on C3): no external pulls that fight the straps, and never drive GPIO12 high at reset on 5 V-tolerant modules.
- Enable the hardware watchdog (esp_task_wdt) around your main loop and use deep-sleep with RTC retention for battery nodes.
- Read the SHT31 status register after each measurement, enable heater only for condensation recovery, and CRC-check every 6-byte frame before trusting values.

# ✅ Combined Production-Grade SHT31 Architecture

```cpp
#include <Wire.h>

#define I2C_SDA 8
#define I2C_SCL 9
#define I2C_FREQ 400000

#define MAX_SENSORS 2

// SHT31 default addresses
uint8_t sensorAddresses[MAX_SENSORS] = {0x44, 0x45};

enum Repeatability {
  REP_HIGH,
  REP_MED,
  REP_LOW
};

struct SHT31 {
  uint8_t address;
  bool connected;
  uint32_t serialNumber;
  float temperature;
  float humidity;
  uint16_t rawTemp;
  uint16_t rawHum;
  bool measurementPending;
  uint32_t measurementStart;
};

SHT31 sensors[MAX_SENSORS];

// ---------------- CRC CHECK ----------------
uint8_t crc8(const uint8_t *data, int len) {
  uint8_t crc = 0xFF;
  for (int i = 0; i < len; i++) {
    crc ^= data[i];
    for (int b = 0; b < 8; b++)
      crc = (crc & 0x80) ? (crc << 1) ^ 0x31 : (crc << 1);
  }
  return crc;
}

// ---------------- LOW LEVEL WRITE ----------------
bool writeCommand(uint8_t addr, uint16_t cmd) {
  Wire.beginTransmission(addr);
  Wire.write(cmd >> 8);
  Wire.write(cmd & 0xFF);
  return (Wire.endTransmission() == 0);
}

// ---------------- CONNECTION CHECK ----------------
bool isConnected(uint8_t addr) {
  Wire.beginTransmission(addr);
  return (Wire.endTransmission() == 0);
}

// ---------------- SERIAL NUMBER ----------------
uint32_t getSerialNumber(uint8_t addr) {
  writeCommand(addr, 0x3780);
  delay(10);

  Wire.requestFrom(addr, (uint8_t)6);
  if (Wire.available() != 6) return 0;

  uint8_t data[6];
  for (int i = 0; i < 6; i++) data[i] = Wire.read();

  if (crc8(data, 2) != data[2]) return 0;
  if (crc8(data + 3, 2) != data[5]) return 0;

  return ((uint32_t)data[0] << 24) |
         ((uint32_t)data[1] << 16) |
         ((uint32_t)data[3] << 8)  |
          data[4];
}

// ---------------- HEATER CONTROL ----------------
void heaterOn(uint8_t addr) {
  writeCommand(addr, 0x306D);
}

void heaterOff(uint8_t addr) {
  writeCommand(addr, 0x3066);
}

// ---------------- START MEASUREMENT (ASYNC) ----------------
void startMeasurement(SHT31 &sensor, Repeatability rep) {

  uint16_t cmd;

  if (rep == REP_HIGH) cmd = 0x2400;
  else if (rep == REP_MED) cmd = 0x240B;
  else cmd = 0x2416;

  writeCommand(sensor.address, cmd);

  sensor.measurementPending = true;
  sensor.measurementStart = millis();
}

// ---------------- READ MEASUREMENT ----------------
bool readMeasurement(SHT31 &sensor) {

  if (!sensor.measurementPending) return false;

  // wait minimum conversion time (~15ms high)
  if (millis() - sensor.measurementStart < 20) return false;

  Wire.requestFrom(sensor.address, (uint8_t)6);
  if (Wire.available() != 6) return false;

  uint8_t data[6];
  for (int i = 0; i < 6; i++) data[i] = Wire.read();

  if (crc8(data, 2) != data[2]) return false;
  if (crc8(data + 3, 2) != data[5]) return false;

  sensor.rawTemp = (data[0] << 8) | data[1];
  sensor.rawHum  = (data[3] << 8) | data[4];

  sensor.temperature = -45 + 175 * ((float)sensor.rawTemp / 65535.0);
  sensor.humidity    = 100 * ((float)sensor.rawHum / 65535.0);

  sensor.measurementPending = false;
  return true;
}

// ---------------- LAST READ ----------------
bool readLastMeasurement(SHT31 &sensor) {
  writeCommand(sensor.address, 0xE000);
  delay(5);
  return readMeasurement(sensor);
}

// ---------------- INIT SENSOR ----------------
void initSensor(SHT31 &sensor, uint8_t addr) {
  sensor.address = addr;
  sensor.connected = isConnected(addr);

  if (sensor.connected) {
    sensor.serialNumber = getSerialNumber(addr);
  }

  sensor.measurementPending = false;
}

// ---------------- SETUP ----------------
void setup() {
  Serial.begin(115200);

  Wire.begin(I2C_SDA, I2C_SCL);
  Wire.setClock(I2C_FREQ);

  for (int i = 0; i < MAX_SENSORS; i++) {
    initSensor(sensors[i], sensorAddresses[i]);
  }
}

// ---------------- LOOP ----------------
void loop() {

  for (int i = 0; i < MAX_SENSORS; i++) {

    if (!sensors[i].connected) continue;

    if (!sensors[i].measurementPending) {
      startMeasurement(sensors[i], REP_HIGH);
    }

    if (readMeasurement(sensors[i])) {

      Serial.print("Sensor ");
      Serial.print(i);
      Serial.print(" | Temp: ");
      Serial.print(sensors[i].temperature);
      Serial.print(" C | Hum: ");
      Serial.print(sensors[i].humidity);
      Serial.println(" %");
    }
  }
}
```
