#include <Wire.h>
#include "MAX30105.h"

namespace {
constexpr uint8_t kLedBrightness = 0x5F;
constexpr uint8_t kSampleAverage = 4;
constexpr uint8_t kLedMode = 3; // RED + IR
constexpr uint8_t kPulseWidth = 411;
constexpr uint16_t kAdcRange = 4096;

constexpr uint16_t kSampleRateHz = 50;
constexpr uint32_t kInitTimeMs = 10000;
constexpr uint32_t kSampleIntervalMs = 1000 / kSampleRateHz;

constexpr uint32_t kIrContactThreshold = 20000;
constexpr uint32_t kIrMotionThreshold = 10000;

constexpr uint16_t kBpmMin = 40;
constexpr uint16_t kBpmMax = 180;

constexpr float kAlpha = 0.90f;
constexpr float kBaselineAlpha = 0.995f;
constexpr uint8_t kWindowSize = 5;
constexpr uint8_t kPrintBatchSize = 3;
constexpr uint32_t kPrintIntervalMs = 333;
} // namespace

MAX30105 sensor;

struct FilterState {
  float irFiltered = 0.0f;
  float redFiltered = 0.0f;
  float irBaseline = 0.0f;
  float redBaseline = 0.0f;
};

struct PeakState {
  float previous = 0.0f;
  bool rising = false;
  unsigned long lastBeatMs = 0;
};

struct BpmAverager {
  float window[kWindowSize] = {0.0f};
  uint8_t windowIndex = 0;
  uint8_t windowCount = 0;

  float prints[kPrintBatchSize] = {0.0f};
  uint8_t printIndex = 0;
  unsigned long lastPrintMs = 0;
};

FilterState filterState;
PeakState peakState;
BpmAverager bpmAverager;

unsigned long initStartMs = 0;
unsigned long lastSampleMs = 0;
uint32_t lastIrRaw = 0;

bool readSample(uint32_t &irRaw, uint32_t &redRaw) {
  if (!sensor.available()) {
    return false;
  }

  irRaw = sensor.getIR();
  redRaw = sensor.getRed();
  sensor.nextSample();
  return true;
}

void updateFilters(uint32_t irRaw, uint32_t redRaw, float &irAc, float &redAc) {
  filterState.irFiltered = kAlpha * filterState.irFiltered + (1.0f - kAlpha) * irRaw;
  filterState.redFiltered = kAlpha * filterState.redFiltered + (1.0f - kAlpha) * redRaw;

  filterState.irBaseline =
      kBaselineAlpha * filterState.irBaseline + (1.0f - kBaselineAlpha) * filterState.irFiltered;
  filterState.redBaseline =
      kBaselineAlpha * filterState.redBaseline + (1.0f - kBaselineAlpha) * filterState.redFiltered;

  irAc = filterState.irFiltered - filterState.irBaseline;
  redAc = filterState.redFiltered - filterState.redBaseline;
}

void printBpmBatch(float bpm) {
  bpmAverager.window[bpmAverager.windowIndex++] = bpm;
  if (bpmAverager.windowIndex >= kWindowSize) {
    bpmAverager.windowIndex = 0;
  }

  if (bpmAverager.windowCount < kWindowSize) {
    bpmAverager.windowCount++;
  }

  float sum = 0.0f;
  for (uint8_t i = 0; i < bpmAverager.windowCount; i++) {
    sum += bpmAverager.window[i];
  }
  const float avg = sum / bpmAverager.windowCount;

  if (millis() - bpmAverager.lastPrintMs < kPrintIntervalMs) {
    return;
  }

  bpmAverager.lastPrintMs = millis();
  bpmAverager.prints[bpmAverager.printIndex++] = avg;

  if (bpmAverager.printIndex >= kPrintBatchSize) {
    Serial.print(bpmAverager.prints[0], 1);
    Serial.print(", ");
    Serial.print(bpmAverager.prints[1], 1);
    Serial.print(", ");
    Serial.println(bpmAverager.prints[2], 1);
    bpmAverager.printIndex = 0;
  }
}

void setupSensor() {
  if (!sensor.begin(Wire)) {
    while (true) {
    }
  }

  sensor.setup(
      kLedBrightness,
      kSampleAverage,
      kLedMode,
      kSampleRateHz,
      kPulseWidth,
      kAdcRange);

  sensor.setPulseAmplitudeIR(0x0F);
  sensor.setPulseAmplitudeRed(0x0F);
}

void setup() {
  Serial.begin(115200);
  Wire.begin();
  setupSensor();
  initStartMs = millis();
}

void loop() {
  if (millis() - initStartMs < kInitTimeMs) {
    return;
  }

  if (millis() - lastSampleMs < kSampleIntervalMs) {
    return;
  }
  lastSampleMs = millis();

  uint32_t irRaw = 0;
  uint32_t redRaw = 0;
  if (!readSample(irRaw, redRaw)) {
    return;
  }

  if (irRaw < kIrContactThreshold) {
    lastIrRaw = irRaw;
    return;
  }

  if (lastIrRaw != 0 && (irRaw > lastIrRaw + kIrMotionThreshold ||
                         irRaw + kIrMotionThreshold < lastIrRaw)) {
    lastIrRaw = irRaw;
    return;
  }
  lastIrRaw = irRaw;

  float irAc = 0.0f;
  float redAc = 0.0f;
  updateFilters(irRaw, redRaw, irAc, redAc);
  (void)redAc;

  if (irAc > peakState.previous && !peakState.rising) {
    peakState.rising = true;
  }

  if (irAc < peakState.previous && peakState.rising) {
    peakState.rising = false;
    const unsigned long now = millis();
    const unsigned long interval = now - peakState.lastBeatMs;

    if (interval > (60000 / kBpmMax) && interval < (60000 / kBpmMin)) {
      const float bpm = 60000.0f / interval;
      printBpmBatch(bpm);
    }

    peakState.lastBeatMs = now;
  }

  peakState.previous = irAc;
}
