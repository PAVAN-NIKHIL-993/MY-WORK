### Suggestions & Next Steps

- Power the ESP32 from a clean 3.3 V rail: ≥100 µF bulk + 100 nF ceramic at the VDD pins, and a separate regulator for servos/motors — brownouts masquerade as random firmware bugs.
- Respect boot strapping pins (GPIO 0/2/12 on classic ESP32, GPIO 2/8/9 on C3): no external pulls that fight the straps, and never drive GPIO12 high at reset on 5 V-tolerant modules.
- Enable the hardware watchdog (esp_task_wdt) around your main loop and use deep-sleep with RTC retention for battery nodes.
- Gate every BPM/SpO₂ output behind a perfusion threshold (MIN_IR_LEVEL) + signal-quality index and a 5–6 s hold; never publish raw beats to UI.
- Calibrate the IR/RED ratio against a reference pulse oximeter at rest and motion to derive your personal SpO₂ curve; ambient-light cancellation (dark current subtraction) is mandatory in daylight.
- Use MQTT/WebSocket with TLS + rotating auth tokens, buffer readings offline (SPIFFS/LittleFS) and replay on reconnect; free tiers cap messages/day — budget before production.

# 🔷 4️⃣ Alert Evaluation Engine

```cpp
Alert evaluateAlerts(float hr, float spo2, float temp, float pi, bool irregularPulse) {

  Alert alert = {NONE, INFO, "Stable"};

  // 🔵 HEART RATE

  if (hr < HR_LOW_THRESHOLD) {
    alert = {LOW_HEART_RATE, WARNING, "Low Heart Rate Alert"};
    return alert;
  }

  if (hr > HR_HIGH_THRESHOLD) {
    alert = {ELEVATED_HEART_RATE, WARNING, "Elevated Heart Rate Alert"};
    return alert;
  }

  if (irregularPulse) {
    alert = {IRREGULAR_PULSE, WARNING, "Irregular Pulse Detected"};
    return alert;
  }

  // 🟢 OXYGEN

  if (spo2 < SPO2_CRITICAL_THRESHOLD) {
    alert = {ACUTE_DESATURATION, CRITICAL, "Acute Desaturation Event"};
    return alert;
  }

  if (spo2 < SPO2_LOW_THRESHOLD) {
    alert = {LOW_OXYGEN, WARNING, "Low Oxygen Saturation"};
    return alert;
  }

  // 🟡 PERFUSION

  if (pi < PI_LOW_THRESHOLD) {
    alert = {LOW_PERFUSION, WARNING, "Low Peripheral Perfusion"};
    return alert;
  }

  // 🟠 TEMPERATURE

  if (temp < TEMP_LOW_THRESHOLD) {
    alert = {LOW_BODY_TEMP, WARNING, "Low Body Temperature"};
    return alert;
  }

  if (temp > TEMP_HIGH_THRESHOLD) {
    alert = {ELEVATED_BODY_TEMP, WARNING, "Elevated Body Temperature"};
    return alert;
  }

  // 🔴 MULTI-PARAMETER INSTABILITY (Early Shock Pattern)
  if (hr > 100 && spo2 < 94 && pi < 0.5) {
    alert = {EARLY_SHOCK_RISK, CRITICAL, "Early Shock Risk"};
    return alert;
  }

  return alert;
}
```

Minimal. Clean. Deterministic.

```cpp
#include <WiFi.h>
#include <AsyncTCP.h>
#include <ESPAsyncWebServer.h>
#include <ArduinoJson.h>
#include <Wire.h>
#include <SPIFFS.h>
#include "MAX30105.h"

MAX30105 sensor;

AsyncWebServer server(80);
AsyncWebSocket ws("/ws");

// ------------ WIFI ------------
const char* ssid = "YOUR_WIFI_NAME";
const char* password = "YOUR_WIFI_PASSWORD";

// ------------ HR SETTINGS ------------
#define IR_THRESHOLD 30000
#define MIN_IBI 300
#define MAX_IBI 1500
#define EMA_ALPHA 0.2

// Alert thresholds
#define HR_LOW_THRESHOLD 50
#define HR_HIGH_THRESHOLD 110
#define SPO2_LOW_THRESHOLD 92
#define SPO2_CRITICAL_THRESHOLD 85
#define TEMP_LOW_THRESHOLD 35.0
#define TEMP_HIGH_THRESHOLD 38.0

float irDC = 0;
float irAC = 0;
float irPrev = 0;
float bpmEMA = 0;
unsigned long lastBeat = 0;

float temperature = 36.8;
int spo2 = 98;
int battery = 85;

// ------------ ALERT SYSTEM ------------

enum AlertSeverity {
  ALERT_INFO,
  ALERT_WARNING,
  ALERT_CRITICAL
};

struct Alert {
  String label;
  AlertSeverity severity;
};

Alert evaluateAlert() {

  Alert alert = {"Stable", ALERT_INFO};

  if (bpmEMA > HR_HIGH_THRESHOLD)
    return {"Elevated Heart Rate Alert", ALERT_WARNING};

  if (bpmEMA > 0 && bpmEMA < HR_LOW_THRESHOLD)
    return {"Low Heart Rate Alert", ALERT_WARNING};

  if (spo2 < SPO2_CRITICAL_THRESHOLD)
    return {"Acute Desaturation Event", ALERT_CRITICAL};

  if (spo2 < SPO2_LOW_THRESHOLD)
    return {"Low Oxygen Saturation", ALERT_WARNING};

  if (temperature < TEMP_LOW_THRESHOLD)
    return {"Low Body Temperature", ALERT_WARNING};

  if (temperature > TEMP_HIGH_THRESHOLD)
    return {"Elevated Body Temperature", ALERT_WARNING};

  if (bpmEMA > 100 && spo2 < 94)
    return {"Early Shock Risk", ALERT_CRITICAL};

  return alert;
}

// ------------ SEND JSON DATA ------------

void sendData() {

  StaticJsonDocument<256> doc;

  Alert currentAlert = evaluateAlert();

  doc["pulse"] = (int)bpmEMA;
  doc["temp"] = temperature;
  doc["spo2"] = spo2;
  doc["battery"] = battery;
  doc["alert_label"] = currentAlert.label;

  switch (currentAlert.severity) {
    case ALERT_INFO: doc["alert_severity"] = "INFO"; break;
    case ALERT_WARNING: doc["alert_severity"] = "WARNING"; break;
    case ALERT_CRITICAL: doc["alert_severity"] = "CRITICAL"; break;
  }

  String json;
  serializeJson(doc, json);
  ws.textAll(json);
}

// ------------ WEBSOCKET ------------

void onWebSocketEvent(AsyncWebSocket *server,
                      AsyncWebSocketClient *client,
                      AwsEventType type,
                      void *arg,
                      uint8_t *data,
                      size_t len) {}

// ------------ SETUP ------------

void setup() {

  Serial.begin(115200);
  Wire.begin();

  if (!SPIFFS.begin(true)) {
    Serial.println("SPIFFS Mount Failed");
    return;
  }

  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }

  Serial.println("\nConnected");
  Serial.println(WiFi.localIP());

  if (!sensor.begin(Wire, I2C_SPEED_FAST)) {
    Serial.println("MAX30102 not found");
    while (1);
  }

  sensor.setup();
  sensor.setPulseAmplitudeRed(0x3F);
  sensor.setPulseAmplitudeIR(0x3F);

  ws.onEvent(onWebSocketEvent);
  server.addHandler(ws);

  server.serveStatic("/", SPIFFS, "/").setDefaultFile("index.html");
  server.begin();
}

// ------------ LOOP ------------

void loop() {

  uint32_t ir = sensor.getIR();

  if (ir > IR_THRESHOLD) {

    irDC = 0.95 * irDC + 0.05 * ir;
    irAC = ir - irDC;

    if (irPrev < 0 && irAC >= 0) {

      unsigned long now = millis();
      unsigned long ibi = now - lastBeat;
      lastBeat = now;

      if (ibi > MIN_IBI && ibi < MAX_IBI) {

        float bpm = 60000.0 / ibi;

        if (bpmEMA == 0)
          bpmEMA = bpm;
        else
          bpmEMA += EMA_ALPHA * (bpm - bpmEMA);
      }
    }

    irPrev = irAC;
  }

  static unsigned long lastSend = 0;

  if (millis() - lastSend > 1000) {
    lastSend = millis();
    sendData();
  }
}
```
