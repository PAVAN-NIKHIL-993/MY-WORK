```cpp
#include <Wire.h>
#include "MAX30105.h"

MAX30105 ppg;

// ===== PARAMETERS =====
#define BUFFER_SIZE 300

float dc_avg = 0;
float lp_out = 0;

float alpha_dc = 0.01;
float alpha_lp = 0.95;

// ===== RR VARIABLES =====
float prev = 0, prev_prev = 0;
float curr = 0;

float diff = 0, prev_diff = 0;

float cycle_min = 999, cycle_max = -999;

unsigned long last_rise_time = 0;

// tuning
float EPS = 0.0005;
float AMP_TH = 0.002;
int MIN_CYCLE = 2000;
int MAX_CYCLE = 10000;

// ===== FINGER DETECTION =====
long FINGER_THRESHOLD = 50000;
unsigned long lastFingerCheck = 0;

void setup() {
  Serial.begin(115200);
  Wire.begin();

  ppg.begin(Wire);
  ppg.setup();
  ppg.setPulseAmplitudeIR(0x1F);
}

void loop() {

  long ir = ppg.getIR();
  unsigned long now = millis();

  // ===== NO FINGER HANDLING (every 1 sec) =====
  if (ir < FINGER_THRESHOLD) {
    if (now - lastFingerCheck >= 1000) {
      Serial.println("No finger");
      lastFingerCheck = now;
    }

    // reset variables
    last_rise_time = 0;
    cycle_min = 999;
    cycle_max = -999;

    return;
  }

  // ===== DC REMOVAL =====
  dc_avg = (1 - alpha_dc) * dc_avg + alpha_dc * ir;
  float ac = ir - dc_avg;

  // ===== NORMALIZATION =====
  float ppg_norm = ac / dc_avg;

  // ===== LOW PASS FILTER =====
  lp_out = alpha_lp * lp_out + (1 - alpha_lp) * ppg_norm;

  // ===== SLOPE METHOD =====
  curr = lp_out;
  diff = (curr - prev_prev) / 2;

  if (abs(diff) < EPS) diff = 0;

  if (curr > cycle_max) cycle_max = curr;
  if (curr < cycle_min) cycle_min = curr;

  // detect inhale start
  if (prev_diff <= 0 && diff > 0) {

    if (last_rise_time != 0) {

      float cycle_time = now - last_rise_time;

      if (cycle_time > MIN_CYCLE && cycle_time < MAX_CYCLE) {

        float amplitude = cycle_max - cycle_min;

        if (amplitude > AMP_TH) {

          float rr = 60000.0 / cycle_time;

          Serial.print("RR: ");
          Serial.println(rr);
        }
      }
    }

    cycle_min = 999;
    cycle_max = -999;

    last_rise_time = now;
  }

  prev_prev = prev;
  prev = curr;
  prev_diff = diff;
}
```
