## 🔬 The Math (Steinhart–Hart)

\[
\frac{1}{T} = A + B \ln(R) + C (\ln(R))^3
\]

- **T** → temperature in **Kelvin**
- **R** → thermistor resistance in ohms  
- **A, B, C** → calibration coefficients

If you actually calibrate with 3 known temperatures, you can hit ±0.1°C. If you don’t calibrate, you’re just hoping physics feels generous.

## 🧾 Full ESP32-C3 Code (Arduino)

```cpp
#define THERMISTOR_PIN 2
#define SERIES_RESISTOR 100000.0
#define ADC_MAX 4095.0
#define VCC 3.3

// Steinhart-Hart coefficients (example values)
#define A 0.001129148
#define B 0.000234125
#define C 0.0000000876741

void setup() {
  Serial.begin(115200);
  analogReadResolution(12);
}

void loop() {
  int adcValue = analogRead(THERMISTOR_PIN);

  // Convert ADC to voltage
  float voltage = (adcValue / ADC_MAX) * VCC;

  // Calculate thermistor resistance
  float resistance = SERIES_RESISTOR * (voltage / (VCC - voltage));

  // Apply Steinhart-Hart
  float lnR = log(resistance);
  float invT = A + B * lnR + C * pow(lnR, 3);
  float temperatureK = 1.0 / invT;
  float temperatureC = temperatureK - 273.15;

  Serial.print("Temp: ");
  Serial.print(temperatureC);
  Serial.println(" °C");

  delay(1000);
}
```

```cpp
#define THERMISTOR_PIN 2
#define SERIES_RESISTOR 100000.0
#define ADC_MAX 4095.0
#define VCC 3.3

// Thermistor parameters
#define BETA 3950.0
#define R0 100000.0
#define T0 298.15  // 25°C in Kelvin

float readFilteredADC() {
  long sum = 0;
  for (int i = 0; i < 50; i++) {
    sum += analogRead(THERMISTOR_PIN);
    delay(2);
  }
  return sum / 50.0;
}

void setup() {
  Serial.begin(115200);
  analogReadResolution(12);
}

void loop() {

  float adcValue = readFilteredADC();

  // Convert ADC to voltage
  float voltage = (adcValue / ADC_MAX) * VCC;

  // Calculate thermistor resistance
  float resistance = SERIES_RESISTOR * (voltage / (VCC - voltage));

  // Beta formula
  float temperatureK = 1.0 / ((1.0 / T0) + (1.0 / BETA) * log(resistance / R0));
  float temperatureC = temperatureK - 273.15;

  Serial.print("Temperature: ");
  Serial.print(temperatureC, 2);
  Serial.println(" °C");

  delay(1000);
}
```

```cpp
  2.20, 2.05, 1.90, 1.75, 1.60, 1.48, 1.35, 1.25, 1.15, 1.05, 0.98

    if (voltage <= voltTable[i] && voltage >= voltTable[i + 1]) {

      float t1 = tempTable[i];
      float t2 = tempTable[i + 1];

      float v1 = voltTable[i];
      float v2 = voltTable[i + 1];

      // Linear interpolation
      return t1 + (voltage - v1) * (t2 - t1) / (v2 - v1);
```
