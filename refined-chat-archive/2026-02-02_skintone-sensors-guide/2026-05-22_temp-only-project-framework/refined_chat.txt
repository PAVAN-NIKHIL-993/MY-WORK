### Suggestions & Next Steps

- Power the ESP32 from a clean 3.3 V rail: ≥100 µF bulk + 100 nF ceramic at the VDD pins, and a separate regulator for servos/motors — brownouts masquerade as random firmware bugs.
- Respect boot strapping pins (GPIO 0/2/12 on classic ESP32, GPIO 2/8/9 on C3): no external pulls that fight the straps, and never drive GPIO12 high at reset on 5 V-tolerant modules.
- Enable the hardware watchdog (esp_task_wdt) around your main loop and use deep-sleep with RTC retention for battery nodes.
- Derive A/B/C Steinhart–Hart coefficients from three known temperature points with a calibrated reference (ice point, room temp, ~50 °C oil bath) rather than trusting datasheet nominal constants.
- Keep excitation current low (<100 µA) and average ADC over ≥16 samples to keep self-heating error below 0.1 °C for 100k NTC @ 25 °C.

```cpp
// =========================================
// NTC 100K TEMPERATURE SENSOR
// NodeMCU ESP8266
// =========================================

#include <math.h>

// -------------------------
// PIN
// -------------------------
#define THERMISTOR_PIN A0

// -------------------------
// NTC SETTINGS
// -------------------------
#define SERIES_RESISTOR 100000   // 100K resistor
#define NOMINAL_RESISTANCE 100000 // NTC 100K
#define NOMINAL_TEMPERATURE 25    // 25°C
#define BETA_COEFFICIENT 3950     // Common Beta value

// -------------------------
// VARIABLES
// -------------------------
float temperatureC = 0;

void setup() {

    Serial.begin(115200);

    Serial.println();
    Serial.println("=================================");
    Serial.println(" NTC 100K TEMPERATURE MONITOR ");
    Serial.println("=================================");
}

void loop() {

    // -------------------------
    // READ ADC
    // -------------------------
    int adcValue = analogRead(THERMISTOR_PIN);

    // Prevent divide-by-zero catastrophe
    if (adcValue == 0) {
        Serial.println("Sensor Error");
        delay(1000);
        return;
    }

    // -------------------------
    // CALCULATE RESISTANCE
    // -------------------------
    float resistance = SERIES_RESISTOR / ((1023.0 / adcValue) - 1);

    // -------------------------
    // STEINHART-HART
    // -------------------------
    float steinhart;

    steinhart = resistance / NOMINAL_RESISTANCE;
    steinhart = log(steinhart);
    steinhart /= BETA_COEFFICIENT;
    steinhart += 1.0 / (NOMINAL_TEMPERATURE + 273.15);
    steinhart = 1.0 / steinhart;
    temperatureC = steinhart - 273.15;

    // -------------------------
    // SERIAL OUTPUT
    // -------------------------
    Serial.println("=================================");

    Serial.print("ADC VALUE: ");
    Serial.println(adcValue);

    Serial.print("RESISTANCE: ");
    Serial.print(resistance);
    Serial.println(" Ohms");

    Serial.print("TEMPERATURE: ");
    Serial.print(temperatureC);
    Serial.println(" °C");

    // -------------------------
    // JSON OUTPUT
    // -------------------------
    String json = "{";
    json += "\"temperature\":";
    json += String(temperatureC, 1);
    json += "}";

    Serial.println(json);

    Serial.println("=================================");

    delay(1000);
}
```

```cpp
// ======================================================
// HIGH ACCURACY NTC 100K TEMPERATURE MONITOR
// NodeMCU ESP8266
// Outputs BOTH Celsius and Fahrenheit
// ======================================================

#include <math.h>

// -------------------------
// PIN
// -------------------------
#define THERMISTOR_PIN A0

// -------------------------
// NTC CONFIGURATION
// -------------------------
#define SERIES_RESISTOR      100000.0   // 100K resistor
#define NOMINAL_RESISTANCE   100000.0   // NTC 100K at 25°C
#define NOMINAL_TEMPERATURE  25.0       // 25°C
#define BETA_COEFFICIENT     3950.0     // Beta value

// -------------------------
// VARIABLES
// -------------------------
float temperatureC = 0;
float temperatureF = 0;

// ======================================================
// SETUP
// ======================================================

void setup() {

    Serial.begin(115200);

    Serial.println();
    Serial.println("======================================");
    Serial.println(" HIGH ACCURACY NTC TEMP MONITOR ");
    Serial.println("======================================");
}

// ======================================================
// LOOP
// ======================================================

void loop() {

    // ----------------------------------
    // MULTIPLE SAMPLES FOR ACCURACY
    // ----------------------------------

    long totalADC = 0;

    for (int i = 0; i < 20; i++) {

        totalADC += analogRead(THERMISTOR_PIN);
        delay(10);
    }

    float adcValue = totalADC / 20.0;

    // ----------------------------------
    // PREVENT INVALID VALUES
    // ----------------------------------

    if (adcValue <= 0 || adcValue >= 1023) {

        Serial.println("Sensor Error");
        delay(1000);
        return;
    }

    // ----------------------------------
    // CALCULATE RESISTANCE
    // ----------------------------------

    float resistance;

    resistance = SERIES_RESISTOR / ((1023.0 / adcValue) - 1.0);

    // ----------------------------------
    // STEINHART-HART EQUATION
    // ----------------------------------

    float steinhart;

    steinhart = resistance / NOMINAL_RESISTANCE;
    steinhart = log(steinhart);
    steinhart /= BETA_COEFFICIENT;
    steinhart += 1.0 / (NOMINAL_TEMPERATURE + 273.15);
    steinhart = 1.0 / steinhart;

    // Celsius
    temperatureC = steinhart - 273.15;

    // Fahrenheit
    temperatureF = (temperatureC * 9.0 / 5.0) + 32.0;

    // ----------------------------------
    // PRETTY SERIAL OUTPUT
    // ----------------------------------

    Serial.println("======================================");

    Serial.print("ADC VALUE      : ");
    Serial.println(adcValue);

    Serial.print("RESISTANCE     : ");
    Serial.print(resistance);
    Serial.println(" Ohms");

    Serial.print("TEMPERATURE C  : ");
    Serial.print(temperatureC, 2);
    Serial.println(" °C");

    Serial.print("TEMPERATURE F  : ");
    Serial.print(temperatureF, 2);
    Serial.println(" °F");

    // ----------------------------------
    // JSON OUTPUT
    // ----------------------------------

    String json = "{";

    json += "\"tempC\":";
    json += String(temperatureC, 2);

    json += ",";

    json += "\"tempF\":";
    json += String(temperatureF, 2);

    json += "}";

    Serial.println(json);

    Serial.println("======================================");

    delay(1000);
}
```
