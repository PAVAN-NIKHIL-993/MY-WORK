```cpp
#include <Wire.h>
#include "MAX30105.h"
#include "heartRate.h"

#include <WiFi.h>
#include <WebSocketsServer.h>

MAX30105 particleSensor;
WebSocketsServer webSocket = WebSocketsServer(81);

// WiFi creds
const char* ssid = "YOUR_WIFI";
const char* password = "YOUR_PASS";

long irValue;
float beatsPerMinute;
int beatAvg;

float spo2 = 0;
float rr = 0;

// buffers
#define BUFFER_SIZE 100
long irBuffer[BUFFER_SIZE];
long redBuffer[BUFFER_SIZE];

int bufferIndex = 0;

unsigned long lastBeat = 0;

// -------------------- SETUP --------------------
void setup() {
  Serial.begin(115200);

  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
  }

  webSocket.begin();

  if (!particleSensor.begin(Wire, I2C_SPEED_FAST)) {
    Serial.println("MAX30102 not found");
    while (1);
  }

  particleSensor.setup();
  particleSensor.setPulseAmplitudeRed(0x0A);
  particleSensor.setPulseAmplitudeIR(0x0A);
}

// -------------------- LOOP --------------------
void loop() {
  webSocket.loop();

  irValue = particleSensor.getIR();

  if (irValue < 50000) return;

  // Store data
  irBuffer[bufferIndex] = particleSensor.getIR();
  redBuffer[bufferIndex] = particleSensor.getRed();
  bufferIndex++;

  if (bufferIndex >= BUFFER_SIZE) {
    bufferIndex = 0;

    // -------- HR --------
    if (checkForBeat(irValue)) {
      long delta = millis() - lastBeat;
      lastBeat = millis();
      beatsPerMinute = 60 / (delta / 1000.0);

      if (beatsPerMinute > 40 && beatsPerMinute < 180) {
        beatAvg = (beatAvg * 0.8) + (beatsPerMinute * 0.2);
      }
    }

    // -------- SpO2 (approx) --------
    float ratio = (float)redBuffer[0] / (float)irBuffer[0];
    spo2 = 110 - 25 * ratio;
    spo2 = constrain(spo2, 90, 100);

    // -------- RR (simple estimation) --------
    rr = estimateRR();

    // -------- SEND JSON --------
    String json = "{";
    json += "\"hr\":" + String(beatAvg) + ",";
    json += "\"spo2\":" + String(spo2) + ",";
    json += "\"rr\":" + String(rr);
    json += "}";

    webSocket.broadcastTXT(json);
  }
}

// -------------------- RR ESTIMATION --------------------
float estimateRR() {
  int crossings = 0;

  for (int i = 1; i < BUFFER_SIZE; i++) {
    if (irBuffer[i] > irBuffer[i - 1]) crossings++;
  }

  float breaths = crossings / 10.0;
  return breaths;
}
```

## 🧠 `.ino` FILE

```cpp
#include <Wire.h>
#include "MAX30105.h"
#include "heartRate.h"

#include <WiFi.h>
#include <WebSocketsServer.h>

MAX30105 particleSensor;
WebSocketsServer webSocket = WebSocketsServer(81);

// ================= WIFI =================
const char* ssid = "YOUR_WIFI_NAME";
const char* password = "YOUR_WIFI_PASSWORD";

// ================= VARIABLES =================
#define BUFFER_SIZE 100

long irBuffer[BUFFER_SIZE];
long redBuffer[BUFFER_SIZE];

int bufferIndex = 0;

float beatsPerMinute;
int beatAvg = 0;

float spo2 = 0;
float rr = 0;

long lastBeat = 0;

// ================= SETUP =================
void setup() {
  Serial.begin(115200);
  delay(1000);

  // WiFi connect
  WiFi.begin(ssid, password);
  Serial.print("Connecting...");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }

  Serial.println("\nConnected!");
  Serial.print("IP: ");
  Serial.println(WiFi.localIP());

  webSocket.begin();

  // Sensor init
  if (!particleSensor.begin(Wire, I2C_SPEED_FAST)) {
    Serial.println("MAX30102 not found!");
    while (1);
  }

  particleSensor.setup();
  particleSensor.setPulseAmplitudeRed(0x0A);
  particleSensor.setPulseAmplitudeIR(0x0A);

  Serial.println("Sensor initialized");
}

// ================= LOOP =================
void loop() {
  webSocket.loop();

  long irValue = particleSensor.getIR();
  long redValue = particleSensor.getRed();

  // Finger detection
  if (irValue < 50000) {
    Serial.println("No finger detected");
    return;
  }

  // Store samples
  irBuffer[bufferIndex] = irValue;
  redBuffer[bufferIndex] = redValue;
  bufferIndex++;

  // -------- HEART RATE --------
  if (checkForBeat(irValue)) {
    long delta = millis() - lastBeat;
    lastBeat = millis();

    beatsPerMinute = 60 / (delta / 1000.0);

    if (beatsPerMinute > 40 && beatsPerMinute < 180) {
      beatAvg = (beatAvg * 0.85) + (beatsPerMinute * 0.15);
    }
  }

  // -------- PROCESS EVERY BUFFER --------
  if (bufferIndex >= BUFFER_SIZE) {
    bufferIndex = 0;

    spo2 = calculateSpO2();
    rr = estimateRR();

    sendData();
  }
}

// ================= SpO2 =================
float calculateSpO2() {
  float irMean = 0, redMean = 0;

  for (int i = 0; i < BUFFER_SIZE; i++) {
    irMean += irBuffer[i];
    redMean += redBuffer[i];
  }

  irMean /= BUFFER_SIZE;
  redMean /= BUFFER_SIZE;

  float ratio = redMean / irMean;

  float spo2 = 110 - 25 * ratio;  // crude formula
  return constrain(spo2, 90, 100);
}

// ================= RR =================
float estimateRR() {
  int peaks = 0;

  for (int i = 1; i < BUFFER_SIZE - 1; i++) {
    if (irBuffer[i] > irBuffer[i - 1] &&
        irBuffer[i] > irBuffer[i + 1]) {
      peaks++;
    }
  }

  float breaths = peaks / 4.0;
  return constrain(breaths, 8, 30);
}

// ================= SEND =================
void sendData() {
  String json = "{";
  json += "\"hr\":" + String(beatAvg) + ",";
  json += "\"spo2\":" + String(spo2) + ",";
  json += "\"rr\":" + String(rr);
  json += "}";

  webSocket.broadcastTXT(json);

  Serial.println(json);
}
```
