#include <Wire.h>
#include "MAX30105.h"

/* =========================================================
   CONFIG
   ========================================================= */
constexpr uint16_t SAMPLE_RATE_HZ = 50;
constexpr uint16_t INIT_TIME_MS = 10000;

constexpr uint32_t IR_CONTACT_TH = 20000;
constexpr uint32_t IR_MOTION_DELTA_TH = 1500;

constexpr uint8_t BPM_MIN = 40;
constexpr uint8_t BPM_MAX = 180;

constexpr float FILTER_ALPHA = 0.90f;
constexpr float BASELINE_ALPHA = 0.995f;
constexpr float PEAK_MIN_AMPLITUDE = 200.0f;

constexpr uint16_t MOTION_HOLDOFF_MS = 500;
constexpr uint8_t WINDOW5_SIZE = 5;
constexpr uint8_t SECOND_BUCKETS = 3;

/* =========================================================
   OBJECT
   ========================================================= */
MAX30105 sensor;

/* =========================================================
   TIMING
   ========================================================= */
unsigned long initStart;
unsigned long lastSampleTime = 0;
unsigned long lastBeatTime = 0;
unsigned long lastMotionTime = 0;

/* =========================================================
   SIGNAL
   ========================================================= */
float irFiltered = 0.0f;
float redFiltered = 0.0f;
float irBaseline = 0.0f;
float redBaseline = 0.0f;
float irAC = 0.0f;
float redAC = 0.0f;
uint32_t lastIrRaw = 0;

/* =========================================================
   AVERAGING
   ========================================================= */
float window5[WINDOW5_SIZE] = {0.0f};
uint8_t w5Index = 0;
uint8_t w5Count = 0;

float secondVals[SECOND_BUCKETS] = {0.0f};
uint8_t secIndex = 0;
unsigned long secTimer = 0;

/* =========================================================
   FORWARD DECLARATIONS
   ========================================================= */
void addBPM(float bpm);
void resetSignalState();

/* =========================================================
   SETUP
   ========================================================= */
void setup() {
  Serial.begin(115200);
  Wire.begin();

  if (!sensor.begin(Wire)) {
    while (1) {
    }
  }

  sensor.setup(
    0x5F,                  // LED brightness
    4,                     // sample average
    3,                     // RED + IR
    SAMPLE_RATE_HZ,        // 50 SPS
    411,                   // pulse width
    4096                   // ADC range
  );

  sensor.setPulseAmplitudeIR(0x0F);
  sensor.setPulseAmplitudeRed(0x0F);

  initStart = millis();
}

/* =========================================================
   LOOP
   ========================================================= */
void loop() {
  /* ---------- 10s INIT ---------- */
  if (millis() - initStart < INIT_TIME_MS) return;

  /* ---------- SAMPLE TIMING ---------- */
  if (millis() - lastSampleTime < (1000 / SAMPLE_RATE_HZ)) return;
  lastSampleTime = millis();

  /* ---------- READ FIFO ---------- */
  if (!sensor.available()) return;

  uint32_t irRaw = sensor.getIR();
  uint32_t redRaw = sensor.getRed();
  sensor.nextSample();

  /* ---------- CONTACT ---------- */
  if (irRaw < IR_CONTACT_TH) {
    resetSignalState();
    return;
  }

  /* ---------- MOTION DETECTION ---------- */
  uint32_t delta = (irRaw > lastIrRaw) ? (irRaw - lastIrRaw) : (lastIrRaw - irRaw);
  lastIrRaw = irRaw;
  if (delta > IR_MOTION_DELTA_TH) {
    lastMotionTime = millis();
  }
  if (millis() - lastMotionTime < MOTION_HOLDOFF_MS) return;

  /* =====================================================
     PRE-PROCESSING
     ===================================================== */
  irFiltered = FILTER_ALPHA * irFiltered + (1.0f - FILTER_ALPHA) * irRaw;
  redFiltered = FILTER_ALPHA * redFiltered + (1.0f - FILTER_ALPHA) * redRaw;

  /* =====================================================
     BASELINE REMOVAL
     ===================================================== */
  irBaseline = BASELINE_ALPHA * irBaseline + (1.0f - BASELINE_ALPHA) * irFiltered;
  redBaseline = BASELINE_ALPHA * redBaseline + (1.0f - BASELINE_ALPHA) * redFiltered;

  irAC = irFiltered - irBaseline;
  redAC = redFiltered - redBaseline;

  /* =====================================================
     PEAK DETECTION (IR)
     ===================================================== */
  static float prev = 0.0f;
  static bool rising = false;

  if (irAC > prev && !rising) rising = true;

  if (irAC < prev && rising) {
    rising = false;

    if (prev > PEAK_MIN_AMPLITUDE) {
      unsigned long now = millis();
      unsigned long interval = now - lastBeatTime;

      if (interval > (60000 / BPM_MAX) && interval < (60000 / BPM_MIN)) {
        float bpm = 60000.0f / interval;
        addBPM(bpm);
      }

      lastBeatTime = now;
    }
  }

  prev = irAC;
}

/* =========================================================
   BPM AVERAGING
   ========================================================= */
void addBPM(float bpm) {
  /* ---- sliding 5 ---- */
  window5[w5Index++] = bpm;
  if (w5Index >= WINDOW5_SIZE) w5Index = 0;
  if (w5Count < WINDOW5_SIZE) w5Count++;

  float sum5 = 0.0f;
  for (uint8_t i = 0; i < w5Count; i++) sum5 += window5[i];
  float avg5 = sum5 / static_cast<float>(w5Count);

  /* ---- 3 prints / second ---- */
  if (millis() - secTimer >= 333) {
    secTimer = millis();
    secondVals[secIndex++] = avg5;

    if (secIndex == SECOND_BUCKETS) {
      Serial.print(secondVals[0], 1);
      Serial.print(", ");
      Serial.print(secondVals[1], 1);
      Serial.print(", ");
      Serial.println(secondVals[2], 1);
      secIndex = 0;
    }
  }
}

/* =========================================================
   RESET SIGNAL STATE
   ========================================================= */
void resetSignalState() {
  irFiltered = 0.0f;
  redFiltered = 0.0f;
  irBaseline = 0.0f;
  redBaseline = 0.0f;
  irAC = 0.0f;
  redAC = 0.0f;
  lastBeatTime = 0;
  lastMotionTime = 0;
  lastIrRaw = 0;
  w5Index = 0;
  w5Count = 0;
  secIndex = 0;
  secTimer = millis();
  for (uint8_t i = 0; i < WINDOW5_SIZE; i++) window5[i] = 0.0f;
  for (uint8_t i = 0; i < SECOND_BUCKETS; i++) secondVals[i] = 0.0f;
}
