#include <SPI.h>
#include <SD.h>
#include "BluetoothA2DPSource.h"

BluetoothA2DPSource a2dp;

File audioFile;

#define SD_CS 5

bool btConnected = false;

uint8_t wavHeader[44];
int16_t audioBuffer[1024]; // 🔥 bigger buffer = smoother audio

// 🔥 YOUR SPEAKER MAC
uint8_t speaker_addr[] = {0x6C, 0xFE, 0x53, 0x52, 0x6A, 0xF2};

// 🎧 Audio callback (continuous streaming)
int32_t get_data(Frame *data, int32_t len) {

  if (!audioFile || !audioFile.available()) {
    memset(data, 0, len); // send silence
    return len;
  }

  int bytesRead = audioFile.read((uint8_t*)audioBuffer, len);

  if (bytesRead <= 0) {
    memset(data, 0, len);
    return len;
  }

  memcpy(data, audioBuffer, bytesRead);
  return bytesRead;
}

// ▶️ Play WAV (blocking until finished)
void playFile(String filename) {

  if (!btConnected) {
    Serial.println("❌ BT not connected");
    return;
  }

  if (audioFile) audioFile.close();

  audioFile = SD.open("/" + filename);

  if (!audioFile) {
    Serial.println("❌ Missing: " + filename);
    return;
  }

  // Skip WAV header
  audioFile.read(wavHeader, 44);

  Serial.println("▶ Playing: " + filename);

  // 🔥 WAIT until file finishes streaming
  while (audioFile.available()) {
    delay(10);
  }

  audioFile.close();

  // small gap between sounds
  delay(100);
}

// 🔢 Speak digits (e.g., 123 → "1 2 3")
void speakDigits(int num) {

  if (num == 0) {
    playFile("0.wav");
    return;
  }

  int digits[10];
  int count = 0;

  while (num > 0) {
    digits[count++] = num % 10;
    num /= 10;
  }

  for (int i = count - 1; i >= 0; i--) {
    playFile(String(digits[i]) + ".wav");
  }
}

// 🔢 Speak full number (e.g., 123 → "one hundred twenty three")
void speakNumber(int num) {

  if (num == 0) {
    playFile("0.wav");
    return;
  }

  if (num >= 100) {
    int hundreds = num / 100;

    playFile(String(hundreds) + ".wav");
    playFile("100.wav");

    num %= 100;
  }

  if (num >= 20) {
    int tens = (num / 10) * 10;

    playFile(String(tens) + ".wav");

    num %= 10;
  }

  if (num > 0 && num < 20) {
    playFile(String(num) + ".wav");
  }
}

void setup() {
  Serial.begin(115200);

  if (!SD.begin(SD_CS)) {
    Serial.println("❌ SD FAIL");
    return;
  }

  Serial.println("✅ SD OK");

  // 🔥 Bluetooth connection callback
  a2dp.set_on_connection_state_changed([](esp_a2d_connection_state_t state, void*) {

    Serial.print("BT State: ");
    Serial.println(state);

    if (state == ESP_A2D_CONNECTION_STATE_CONNECTED) {
      btConnected = true;
      Serial.println("✅ CONNECTED");
    } else {
      btConnected = false;
    }

  });

  // Start A2DP streaming
  a2dp.start("ESP32_SPEAKER", get_data);

  delay(4000);  // allow BT stack to settle

  Serial.println("🔄 Trying MAC connection...");
  a2dp.connect_to(speaker_addr);
}

void loop() {

  if (Serial.available()) {

    int number = Serial.parseInt();

    Serial.print("Input: ");
    Serial.println(number);

    if (!btConnected) {
      Serial.println("⚠️ Not connected yet");
      return;
    }

    // 🔊 First: digit by digit
    speakDigits(number);

    delay(500);

    // 🔊 Then: full number
    speakNumber(number);
  }
}

#include "BluetoothA2DPSource.h"

BluetoothA2DPSource a2dp;

File audioFile;

#define SD_CS 5

bool btConnected = false;
bool isPlaying = false;

uint8_t wavHeader[44];
int16_t audioBuffer[1024]; // bigger buffer = smoother

// 🔥 YOUR SPEAKER MAC
uint8_t speaker_addr[] = {0x6C, 0xFE, 0x53, 0x52, 0x6A, 0xF2};

// 🎧 Audio callback (runs continuously in background)
int32_t get_data(Frame *data, int32_t len) {

  if (!audioFile || !audioFile.available()) {
    isPlaying = false;
    memset(data, 0, len); // send silence
    return len;
  }

  int bytesRead = audioFile.read((uint8_t*)audioBuffer, len);

  if (bytesRead <= 0) {
    isPlaying = false;
    memset(data, 0, len);
    return len;
  }

  memcpy(data, audioBuffer, bytesRead);
  return bytesRead;
}

// ▶️ Start playing file (non-blocking)
void playFile(String filename) {

  if (!btConnected) {
    Serial.println("❌ BT not connected");
    return;
  }

  if (audioFile) audioFile.close();

  audioFile = SD.open("/" + filename);

  if (!audioFile) {
    Serial.println("❌ Missing: " + filename);
    return;
  }

  audioFile.read(wavHeader, 44);

  Serial.println("▶ Playing: " + filename);

  isPlaying = true;
}

// ⏳ Wait until current audio finishes (safe, non-blocking-ish)
void waitUntilDone() {
  while (isPlaying) {
    delay(1); // yield CPU to BT
  }
}

// 🔢 Speak digits
void speakDigits(int num) {

  if (num == 0) {
    playFile("0.wav");
    waitUntilDone();
    return;
  }

  int digits[10];
  int count = 0;

  while (num > 0) {
    digits[count++] = num % 10;
    num /= 10;
  }

  for (int i = count - 1; i >= 0; i--) {
    playFile(String(digits[i]) + ".wav");
    waitUntilDone();
  }
}

// 🔢 Speak full number
void speakNumber(int num) {

  if (num == 0) {
    playFile("0.wav");
    waitUntilDone();
    return;
  }

  if (num >= 100) {
    int hundreds = num / 100;

    playFile(String(hundreds) + ".wav");
    waitUntilDone();

    playFile("100.wav");
    waitUntilDone();

    num %= 100;
  }

  if (num >= 20) {
    int tens = (num / 10) * 10;

    playFile(String(tens) + ".wav");
    waitUntilDone();

    num %= 10;
  }

  if (num > 0 && num < 20) {
    playFile(String(num) + ".wav");
    waitUntilDone();
  }
}

void setup() {
  Serial.begin(115200);

  if (!SD.begin(SD_CS)) {
    Serial.println("❌ SD FAIL");
    return;
  }

  Serial.println("✅ SD OK");

  // 🔵 Bluetooth state callback
  a2dp.set_on_connection_state_changed([](esp_a2d_connection_state_t state, void*) {

    Serial.print("BT State: ");
    Serial.println(state);

    if (state == ESP_A2D_CONNECTION_STATE_CONNECTED) {
      btConnected = true;
      Serial.println("✅ CONNECTED");
    } else {
      btConnected = false;
    }

  });

  // Start A2DP
  a2dp.start("ESP32_SPEAKER", get_data);

  delay(4000); // allow BT stack to stabilize

  Serial.println("🔄 Trying MAC connection...");
  a2dp.connect_to(speaker_addr);
}

void loop() {

  if (Serial.available()) {

    int number = Serial.parseInt();

    Serial.print("Input: ");
    Serial.println(number);

    if (!btConnected) {
      Serial.println("⚠️ Not connected yet");
      return;
    }

    // 🔊 Speak digits
    speakDigits(number);

    delay(200);

    // 🔊 Speak full number
    speakNumber(number);
  }
}
