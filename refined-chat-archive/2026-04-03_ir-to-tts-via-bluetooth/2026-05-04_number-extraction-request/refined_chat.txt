## 🔧 Clean Upgrade (Drop-in Fix)

Add this function:

```cpp
bool isAllowed(String word) {
    word.toUpperCase();

    // Digits 0–9
    if (word.length() == 1 && isDigit(word[0])) return true;

    // 11–19
    int num = word.toInt();
    if (num >= 11 && num <= 19) return true;

    // Allowed words
    const char* allowedWords[] = {
        "TEN","TWENTY","THIRTY","FOURTY","FIFTY",
        "SIXTY","SEVENTY","EIGHTY","NINETY","HUNDRED",
        "TOKEN","NUMBER"
    };

    for (auto &w : allowedWords) {
        if (word == w) return true;
    }

    return false;
}
```
