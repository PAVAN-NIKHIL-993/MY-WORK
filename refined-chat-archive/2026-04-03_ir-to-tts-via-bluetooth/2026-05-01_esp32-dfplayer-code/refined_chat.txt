```cpp
#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include <Keypad.h>
#include "HardwareSerial.h"
#include "DFRobotDFPlayerMini.h"

LiquidCrystal_I2C lcd(0x27, 16, 2);

HardwareSerial mySerial(1);
DFRobotDFPlayerMini dfplayer;

// ================= KEYPAD =================
const byte ROWS = 4;
const byte COLS = 4;

char keys[ROWS][COLS] = {
  {'1','2','3','A'},
  {'4','5','6','B'},
  {'7','8','9','C'},
  {'*','0','#','D'}
};

byte rowPins[ROWS] = {32, 33, 25, 26};
byte colPins[COLS] = {27, 14, 12, 13};

Keypad keypad = Keypad(makeKeymap(keys), rowPins, colPins, ROWS, COLS);

// ================= 7 SEG =================
int digitPins[] = {18, 5, 17, 16};
int segmentPins[] = {4, 2, 15, 13, 12, 14, 27, 26};

byte digits[10] = {
  B00111111, //0
  B00000110, //1
  B01011011, //2
  B01001111, //3
  B01100110, //4
  B01101101, //5
  B01111101, //6
  B00000111, //7
  B01111111, //8
  B01101111  //9
};

int number = 0;

// ================= SETUP =================
void setup() {
  Serial.begin(115200);

  lcd.init();
  lcd.backlight();

  // DFPlayer
  mySerial.begin(9600, SERIAL_8N1, 27, 26);
  if (!dfplayer.begin(mySerial)) {
    Serial.println("DFPlayer ERROR");
  } else {
    dfplayer.volume(25);
  }

  // 7-seg pins
  for (int i = 0; i < 4; i++) pinMode(digitPins[i], OUTPUT);
  for (int i = 0; i < 8; i++) pinMode(segmentPins[i], OUTPUT);

  lcd.setCursor(0,0);
  lcd.print("Enter Number:");
}

// ================= DISPLAY =================
void displayDigit(int digit, int num) {
  digitalWrite(digitPins[digit], LOW);

  byte pattern = digits[num];

  for (int i = 0; i < 8; i++) {
    digitalWrite(segmentPins[i], pattern & (1 << i));
  }

  delay(2);
  digitalWrite(digitPins[digit], HIGH);
}

void displayNumber(int num) {
  int d[4];
  d[0] = (num / 1000) % 10;
  d[1] = (num / 100) % 10;
  d[2] = (num / 10) % 10;
  d[3] = num % 10;

  for (int i = 0; i < 4; i++) {
    displayDigit(i, d[i]);
  }
}

// ================= LOOP =================
void loop() {

  char key = keypad.getKey();

  if (key) {
    if (key >= '0' && key <= '9') {
      number = number * 10 + (key - '0');

      lcd.clear();
      lcd.print("Number:");
      lcd.setCursor(0,1);
      lcd.print(number);
    }

    if (key == '#') {
      dfplayer.play(number);  // plays track = number
    }

    if (key == '*') {
      number = 0;
      lcd.clear();
      lcd.print("Reset");
      delay(500);
      lcd.clear();
      lcd.print("Enter Number:");
    }
  }

  displayNumber(number);
}
```
