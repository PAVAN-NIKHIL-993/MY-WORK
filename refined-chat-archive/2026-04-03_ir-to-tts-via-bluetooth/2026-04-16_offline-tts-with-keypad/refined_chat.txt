## 💻 Full Arduino Code (clean, no nonsense)

```cpp
#include <Keypad.h>
#include <SoftwareSerial.h>
#include <DFRobotDFPlayerMini.h>

SoftwareSerial mySerial(10, 11); // RX, TX
DFRobotDFPlayerMini dfp;

// Keypad setup
const byte ROWS = 4;
const byte COLS = 4;

char keys[ROWS][COLS] = {
  {'1','2','3','A'},
  {'4','5','6','B'},
  {'7','8','9','C'},
  {'*','0','#','D'}
};

byte rowPins[ROWS] = {2,3,4,5};
byte colPins[COLS] = {6,7,8,9};

Keypad keypad = Keypad(makeKeymap(keys), rowPins, colPins, ROWS, COLS);

// Variables
String input = "";
int lastNumber = -1;

void setup() {
  Serial.begin(9600);
  mySerial.begin(9600);

  if (!dfp.begin(mySerial)) {
    Serial.println("DFPlayer not found");
    while(true);
  }

  dfp.volume(25);
  Serial.println("Ready");
}

void loop() {
  char key = keypad.getKey();

  if (key) {
    Serial.print("Key: ");
    Serial.println(key);

    if (key >= '0' && key <= '9') {
      if (input.length() < 3) {
        input += key;
      }
    }

    else if (key == '*') {
      input = "";
      Serial.println("Cleared");
    }

    else if (key == '#') {
      if (input.length() > 0) {
        int number = input.toInt();
        speakNumber(number);
        lastNumber = number;
        input = "";
      }
      else if (lastNumber != -1) {
        speakNumber(lastNumber);
      }
    }
  }
}

// Function to speak number
void speakNumber(int num) {
  Serial.print("Speaking: ");
  Serial.println(num);

  if (num == 0) {
    dfp.play(1); // zero
    return;
  }

  if (num >= 100) {
    int hundreds = num / 100;
    dfp.play(hundreds + 1); // 1-9
    delay(500);
    dfp.play(100); // "hundred"
    delay(500);
    num = num % 100;
  }

  if (num >= 20) {
    int tens = (num / 10) * 10;
    dfp.play(tens); // 20,30...
    delay(500);
    num = num % 10;
  }

  if (num > 0 && num < 20) {
    dfp.play(num + 1);
    delay(500);
  }
}
```

#include <Keypad.h>
#include <DFRobotDFPlayerMini.h>

// ================= UART =================
HardwareSerial mySerial(2); // UART2
DFRobotDFPlayerMini dfp;

// ================= KEYPAD =================
const byte ROWS = 4;
const byte COLS = 4;

char keys[ROWS][COLS] = {
  {'1','2','3','A'},
  {'4','5','6','B'},
  {'7','8','9','C'},
  {'*','0','#','D'}
};

byte rowPins[ROWS] = {13, 12, 14, 27};
byte colPins[COLS] = {26, 25, 33, 32};

Keypad keypad = Keypad(makeKeymap(keys), rowPins, colPins, ROWS, COLS);

// ================= VARIABLES =================
String input = "";
int lastNumber = -1;

// ================= SETUP =================
void setup() {
  Serial.begin(115200);

  // UART2 init: RX=16, TX=17
  mySerial.begin(9600, SERIAL_8N1, 16, 17);

  if (!dfp.begin(mySerial)) {
    Serial.println("DFPlayer not detected");
    while (true);
  }

  dfp.volume(25); // 0–30
  Serial.println("System Ready");
}

// ================= LOOP =================
void loop() {
  char key = keypad.getKey();

  if (key) {
    Serial.print("Key: ");
    Serial.println(key);

    // ===== DIGITS =====
    if (key >= '0' && key <= '9') {
      if (input.length() < 3) {
        input += key;
        Serial.print("Input: ");
        Serial.println(input);
      }
    }

    // ===== CLEAR =====
    else if (key == '*') {
      input = "";
      Serial.println("Input Cleared");
    }

    // ===== CONFIRM / REPEAT =====
    else if (key == '#') {

      // If new input exists → speak it
      if (input.length() > 0) {
        int number = input.toInt();
        speakNumber(number);
        lastNumber = number;
        input = "";
      }

      // If no input → repeat last
      else if (lastNumber != -1) {
        speakNumber(lastNumber);
      }
    }
  }
}

// ================= SPEAK FUNCTION =================
void speakNumber(int num) {
  Serial.print("Speaking: ");
  Serial.println(num);

  if (num == 0) {
    dfp.play(1); // zero
    delay(600);
    return;
  }

  // ===== HUNDREDS =====
  if (num >= 100) {
    int hundreds = num / 100;
    dfp.play(hundreds + 1); // 1–9
    delay(600);

    dfp.play(100); // "hundred"
    delay(600);

    num = num % 100;
  }

  // ===== TENS =====
  if (num >= 20) {
    int tens = (num / 10) * 10;
    dfp.play(tens); // 20,30...
    delay(600);

    num = num % 10;
  }

  // ===== 1–19 =====
  if (num > 0 && num < 20) {
    dfp.play(num + 1);
    delay(600);
  }
}
