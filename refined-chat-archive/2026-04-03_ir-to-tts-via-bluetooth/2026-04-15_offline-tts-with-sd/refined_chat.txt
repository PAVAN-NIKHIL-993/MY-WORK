## 🔧 Modified Code (Offline Playback via SD)

You’ll need:
- SD card module
- Same I2S setup
- No WiFi (finally some peace)

```cpp
#include <SPI.h>
#include <SD.h>
#include "Audio.h"

#define SD_CS     5

#define I2S_BCLK 26
#define I2S_LRC  25
#define I2S_DOUT 22

Audio audio;

void setup() {
  Serial.begin(115200);

  // SD init
  if (!SD.begin(SD_CS)) {
    Serial.println("SD Card Failed");
    return;
  }
  Serial.println("SD Card Ready");

  audio.setPinout(I2S_BCLK, I2S_LRC, I2S_DOUT);
  audio.setVolume(8);

  Serial.println("System Ready");
}

void loop() {
  audio.loop();

  if (Serial.available()) {
    String token = Serial.readStringUntil('\n');
    token.trim();

    if (token.length() > 0) {
      Serial.println("Playing token: " + token);
      playToken(token);
    }
  }
}

void playToken(String num) {

  // Example: "5" → "/5.mp3"
  String filePath = "/" + num + ".mp3";

  if (SD.exists(filePath)) {
    audio.connecttoFS(SD, filePath.c_str());

    while (audio.isRunning()) {
      audio.loop();
    }
  } else {
    Serial.println("File not found: " + filePath);
  }
}
```

No WiFi. No drama. Just deterministic behavior like a civilized system.

```cpp
#include <SPI.h>
#include <SD.h>
#include "Audio.h"

#define SD_CS     5

#define I2S_BCLK 26
#define I2S_LRC  25
#define I2S_DOUT 22

Audio audio;

void setup() {
  Serial.begin(115200);

  if (!SD.begin(SD_CS)) {
    Serial.println("SD Card Failed");
    while (1);
  }

  Serial.println("SD Ready");

  audio.setPinout(I2S_BCLK, I2S_LRC, I2S_DOUT);
  audio.setVolume(8);
}

void loop() {
  audio.loop();

  if (Serial.available()) {
    int num = Serial.parseInt();

    if (num >= 0 && num <= 10) {
      playNumber(num);
    } else {
      Serial.println("Enter 0 to 10 only");
    }
  }
}

void playNumber(int num) {
  String path = "/" + String(num) + ".mp3";

  if (SD.exists(path)) {
    Serial.println("Playing: " + path);

    audio.connecttoFS(SD, path.c_str());

    while (audio.isRunning()) {
      audio.loop();
    }
  } else {
    Serial.println("Missing file: " + path);
  }
}
```

```cpp
#include <SPI.h>
#include <SD.h>
#include "Audio.h"

#define SD_CS     5

#define I2S_BCLK 26
#define I2S_LRC  25
#define I2S_DOUT 22

Audio audio;

void setup() {
  Serial.begin(115200);

  if (!SD.begin(SD_CS)) {
    Serial.println("SD Failed");
    while (1);
  }

  Serial.println("SD OK");

  audio.setPinout(I2S_BCLK, I2S_LRC, I2S_DOUT);
  audio.setVolume(8);

  // smaller buffer = safer compile
  audio.setBufferSize(1024, 1024);
}

void loop() {
  audio.loop();

  if (Serial.available()) {
    int num = Serial.parseInt();

    if (num >= 0 && num <= 10) {
      playWav(num);
    } else {
      Serial.println("0-10 only");
    }
  }
}

void playWav(int num) {
  String path = "/" + String(num) + ".wav";

  if (SD.exists(path)) {
    Serial.println("Playing: " + path);

    audio.connecttoFS(SD, path.c_str());

    while (audio.isRunning()) {
      audio.loop();
    }
  } else {
    Serial.println("Missing: " + path);
  }
}
```
