void allOff() {
  for (int i = 0; i < 4; i++) {
    digitalWrite(stages[i], LOW);
  }
}

void setup() {
  Serial.begin(115200);
  for (int i = 0; i < 4; i++) {
    pinMode(stages[i], OUTPUT);
    digitalWrite(stages[i], LOW);
  }
  Serial.println("ESP32 Voltage Selector Ready");
  Serial.println("Commands: 1–4 = ON, 0 = ALL OFF");
}

void loop() {
  if (Serial.available()) {
    char cmd = Serial.read();

    if (cmd < '0' || cmd > '4') return;

    allOff();

    if (cmd == '0') {
      Serial.println("All stages OFF");
    }
    else if (cmd == '1') {
      digitalWrite(stages[0], HIGH);
      Serial.println("Stage 1 ON → 3.0V");
    }
    else if (cmd == '2') {
      digitalWrite(stages[1], HIGH);
      Serial.println("Stage 2 ON → 3.75V");
    }
    else if (cmd == '3') {
      digitalWrite(stages[2], HIGH);
      Serial.println("Stage 3 ON → 4.1V");
    }
    else if (cmd == '4') {
      digitalWrite(stages[3], HIGH);
      Serial.println("Stage 4 ON → 4.95V");
    }
  }
}
